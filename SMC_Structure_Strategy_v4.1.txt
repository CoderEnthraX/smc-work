var array<line>  mainLn = array.new<line>()
var array<label> mainLb = array.new<label>()
var array<int>   mainCnt = array.new<int>()

f_addLn(array<line> la, array<int> cnt, line l) =>
    array.push(la, l)
    array.push(cnt, 0)

f_addLb(array<label> lb, array<int> cnt, label l) =>
    if array.size(cnt) > 0
        array.push(lb, l)
        int k = array.size(cnt) - 1
        array.set(cnt, k, array.get(cnt, k) + 1)

f_cutOld(array<line> la, array<label> lb, array<int> cnt, int cap) =>
    while array.size(la) > cap
        line.delete(array.shift(la))
        int c = array.shift(cnt)
        while c > 0 and array.size(lb) > 0
            label.delete(array.shift(lb))
            c -= 1

f_cutAll(array<line> la, array<label> lb, array<int> cnt) =>
    while array.size(la) > 0
        line.delete(array.pop(la))
    while array.size(lb) > 0
        label.delete(array.pop(lb))
    array.clear(cnt)

method repos(SLine s, int endBar) =>
    if not na(s) and not na(s.ln) and not na(s.lb)
        string th = hAlignOf(s.sid, not s.done)
        if th != "Left"
            s.lb.set_x(f_xClamp(th == "Right" ? endBar : int(math.avg(s.x1, endBar))))

method freeze(SLine s) =>
    if not na(s) and not na(s.ln)
        s.done := true
        [lc, tc, ls, w, txt, ts, tv, th, showT] = styleOf(s.sid, false)
        s.ln.set_x2(bar_index)
        s.ln.set_extend(extend.none)
        s.ln.set_color(lc)
        s.ln.set_width(w)
        s.ln.set_style(lstyle(ls))
        if not na(s.lb)
            s.lb.delete()
            s.lb := na
        if showT
            int lx = f_xClamp(th == "Left" ? s.x1 : th == "Right" ? bar_index : int(math.avg(s.x1, bar_index)))
            s.lb := label.new(lx, s.price, txt, color = color(na), textcolor = tc, style = lblStyle(tv), size = tsize(ts))
        f_addLn(mainLn, mainCnt, s.ln)
        if not na(s.lb)
            f_addLb(mainLb, mainCnt, s.lb)

method drop(SLine s) =>
    if not na(s) and not na(s.ln)
        s.ln.delete()
        if not na(s.lb)
            s.lb.delete()

mkSLine(float p, int b, int sid, bool draw) =>
    [lc, tc, ls, w, txt, ts, tv, th, showT] = styleOf(sid, true)
    line ln = na
    label lb = na
    int _bx = f_xClamp(b)
    if draw

    SLine.new(ln, lb, b, p, sid, bar_index, false, na, na, na, na, na, false, false)

type SwingTracker
    bool  isHigh
    float potential
    int   potentialBar
    int   count
    float refLevel
    bool  swept

method update(SwingTracker t, float swpLvl) =>
    float confP = na
    int   confB = na
    bool bearC  = close < open
    bool bullC  = close > open
    float _pbBd = math.abs(close - open)
    float _pbWk = math.max(high - math.max(open, close), math.min(open, close) - low)
    bool  _pbQ  = not (pbBodyOn or pbBodyRef) or _pbBd > _pbWk
    bool pb     = (t.isHigh ? bearC : bullC) and _pbQ
    float _pbRf = pbBodyRef ? (t.isHigh ? math.min(open, close) : math.max(open, close)) : (t.isHigh ? low : high)
    bool newExt = na(t.potential) or (t.isHigh ? high >= t.potential : low <= t.potential)
    if newExt
        t.potential    := t.isHigh ? high : low
        t.potentialBar := bar_index
        t.count        := pb ? 1 : 0
        t.refLevel     := pb ? _pbRf : na
        t.swept        := false
    else
        if not na(swpLvl) and (t.isHigh ? low < swpLvl : high > swpLvl)
            t.swept := true
        if pb
            if t.count == 0
                t.count    := 1
                t.refLevel := _pbRf
            else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel) and t.swept
                confP := t.potential
                confB := t.potentialBar
                t.potential    := t.isHigh ? high : low
                t.potentialBar := bar_index
                t.count        := 1
                t.refLevel     := _pbRf
                t.swept        := false
            else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
                t.count    := 2
                t.refLevel := _pbRf
            else if t.count == 2 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
                confP := t.potential
                confB := t.potentialBar
                t.potential    := t.isHigh ? high : low
                t.potentialBar := bar_index
                t.count        := 1
                t.refLevel     := _pbRf
                t.swept        := false
    [confP, confB]

var array<float> bufHi = array.new_float()
var array<float> bufLo = array.new_float()
var array<int>   bufDir = array.new_int()

f_retExtreme(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd > 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd - 1))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd - 1
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

f_retExtInc(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd >= 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

method rearm(SwingTracker t, float p, int b) =>
    t.potential    := p
    t.potentialBar := b
    int  sz  = array.size(bufHi)
    int  idx = b - bar_index + sz - 1
    bool ok  = not na(b) and idx >= 0 and idx <= sz - 1
    int  dir = ok ? array.get(bufDir, idx) : 0
    bool pb  = t.isHigh ? dir == -1 : dir == 1
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? array.get(bufLo, idx) : array.get(bufHi, idx)) : na
    t.swept    := false

method armHere(SwingTracker t) =>
    t.potential    := t.isHigh ? high : low
    t.potentialBar := bar_index
    bool pb = t.isHigh ? close < open : close > open
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? low : high) : na
    t.swept    := false

f_isSwing(int pb, bool isHigh, int k) =>
    int  sz  = array.size(bufHi)
    int  idx = pb - bar_index + sz - 1
    bool ok  = not na(pb) and idx - k >= 0 and idx <= sz - 1
    if ok
        float p = isHigh ? array.get(bufHi, idx) : array.get(bufLo, idx)
        for j = 1 to k
            float v = isHigh ? array.get(bufHi, idx - j) : array.get(bufLo, idx - j)
            if isHigh ? v > p : v < p
                ok := false
    ok

f_confExt(array<int> arrB, array<float> arrP, bool wantLow, int afterBar, float beyond) =>
    float p = na
    int   b = na
    int   n = array.size(arrB)
    if n > 0
        for i = n - 1 to 0
            int cb = array.get(arrB, i)
            if cb <= afterBar
                break
            float cp = array.get(arrP, i)
            if (wantLow ? cp <= beyond : cp >= beyond) and (na(p) or (wantLow ? cp < p : cp > p))
                p := cp
                b := cb
    [p, b]

f_swpConf(array<int> arrB, int swpB) =>
    bool ok = false
    int  n  = array.size(arrB)
    if not na(swpB) and n > 0
        for i = n - 1 to 0
            int cb = array.get(arrB, i)
            if cb == swpB
                ok := true
                break
            if cb < swpB
                break
    ok

f_lastMinor(bool wantLow, int beforeBar) =>
    arrB = wantLow ? msLoB : msHiB
    arrP = wantLow ? msLoP : msHiP
    float p = na
    int   b = na
    int   n = array.size(arrB)
    if n > 0
        for i = n - 1 to 0
            if array.get(arrB, i) < beforeBar
                p := array.get(arrP, i)
                b := array.get(arrB, i)
                break
    [p, b]

method armIdm(SLine s) =>
    if not na(s)
        bool isCeil = s.sid == 0 or s.sid == 2
        [ip, ib] = f_lastMinor(isCeil, s.x1)
        s.idmP   := ip
        s.idmB   := ib
        s.idmHit := false
        if not na(ip)
            [xp, xb] = f_retExtreme(s.x1, not isCeil)
            bool takenPast = not na(xp) and (isCeil ? xp < ip : xp > ip)
            bool takenNow  = isCeil ? low < ip : high > ip
            if takenPast or takenNow
                s.idmHit := true
                f_idmMark(s)

f_intArmIdm(int pivB, bool isCeil) =>
    [ip, ib] = f_lastMinor(isCeil, pivB)

    if not na(ip)

        bool takenPast = not na(xp) and (isCeil ? xp < ip : xp > ip)
        bool takenNow  = isCeil ? low < ip : high > ip
        hit := takenPast or takenNow
    [ip, hit]

    syminfo.pointvalue <= 0 ? 1.0 : syminfo.pointvalue

float stCmLots = stCmMode == "Per lot - I set the contract size" ? stCmUnit : _cmAuto

// ROUND-TURN cost for ONE unit, in account currency: commission + the broker's spread
f_stCmU(float ent) =>
    float c = 0.0
    if stCmMode == "Auto - per lot, contract size from the ticker" or stCmMode == "Per lot - I set the contract size"
        c := stCmLots > 0 ? 2.0 * stCmLot / stCmLots : 0.0
    else if stCmMode == "Percent of trade value"
        c := 2.0 * stCmPct / 100.0 * math.max(nz(ent), 0.0)
    c + (stSprd > 0 ? strategy.convert_to_account(stSprd * f_stPv()) : 0.0)

// ROUND-TURN commission that does NOT scale with size, in account currency
f_stCmF() =>
    stCmMode == "Fixed cash per order" ? 2.0 * stCmFix : 0.0

f_stQty(float ent, float rsk, float amt) =>
    float q    = 0.0
    float _per = 0.0
    if stSize == "Fixed quantity"
        q := stQty
    else if stSize == "Fixed cash per trade"
        q := ent > 0 ? stCash / ent : 0.0
    else
        _per := strategy.convert_to_account(rsk * f_stPv()) + f_stCmU(ent)
        float _net = math.max(0.0, amt - f_stCmF())
        q := _per > 0 ? _net / _per : 0.0
    float _st  = stLotStep > 0 and stCmLots > 0 ? stLotStep * stCmLots : 0.0
    float _raw = q
    if _st > 0
        q := math.round(_raw / _st) * _st
        if stRndMax > 0 and _per > 0 and amt > 0 and q * _per + f_stCmF() > amt * (1.0 + stRndMax / 100.0)
            q := math.floor(_raw / _st) * _st
    float _lim = ent > 0 and stLev > 0 ? strategy.equity * stLev / ent : 0.0
    bool  _cap = _lim > 0 and q > _lim
    if _cap
        q := _st > 0 ? math.floor(_lim / _st) * _st : _lim
    [q, _cap]

float _stPvA = strategy.convert_to_account(f_stPv())
float stCmU  = f_stCmU(close)
float stCmPx = stCmTgt and _stPvA > 0 ? stCmU * (stRR + 1.0) / _stPvA : 0.0
