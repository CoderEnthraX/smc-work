f_addLb(array<label> lb, array<int> cnt, label l) =>
    if array.size(cnt) > 0
        array.push(lb, l)
        int k = array.size(cnt) - 1
        array.set(cnt, k, array.get(cnt, k) + 1)

f_cutOld(array<line> la, array<label> lb, array<int> cnt, int cap) =>
    while array.size(la) > cap
        line.delete(array.shift(la))
        int c = array.shift(cnt)
        while c > 0 and array.size(lb) > 0
            label.delete(array.shift(lb))
            c -= 1

f_cutAll(array<line> la, array<label> lb, array<int> cnt) =>
    while array.size(la) > 0
        line.delete(array.pop(la))
    while array.size(lb) > 0
        label.delete(array.pop(lb))
    array.clear(cnt)

hAlignOf(int sid, bool pend) =>
    if pend
        sid <= 1 ? p1_th : p2_th
    else
        switch sid
            0 => b1_th
            1 => b2_th
            2 => c1_th
            => c2_th

method repos(SLine s, int endBar) =>
    if not na(s) and not na(s.ln) and not na(s.lb)
        string th = hAlignOf(s.sid, not s.done)
        if th != "Left"
            s.lb.set_x(f_xClamp(th == "Right" ? endBar : int(math.avg(s.x1, endBar))))

method freeze(SLine s) =>
    if not na(s) and not na(s.ln)
        s.done := true
        [lc, tc, ls, w, txt, ts, tv, th, showT] = styleOf(s.sid, false)
        s.ln.set_x2(bar_index)
        s.ln.set_extend(extend.none)
        s.ln.set_color(lc)
        s.ln.set_width(w)
        s.ln.set_style(lstyle(ls))
        if not na(s.lb)
            s.lb.delete()
            s.lb := na
        if showT
            int lx = f_xClamp(th == "Left" ? s.x1 : th == "Right" ? bar_index : int(math.avg(s.x1, bar_index)))
            s.lb := label.new(lx, s.price, txt, color = color(na), textcolor = tc, style = lblStyle(tv), size = tsize(ts))
        f_addLn(mainLn, mainCnt, s.ln)
        if not na(s.lb)
            f_addLb(mainLb, mainCnt, s.lb)

method drop(SLine s) =>
    if not na(s) and not na(s.ln)
        s.ln.delete()
        if not na(s.lb)
            s.lb.delete()

type SwingTracker
    bool  isHigh
    float potential
    int   potentialBar
    int   count
    float refLevel
    bool  swept

method update(SwingTracker t, float swpLvl) =>
    float confP = na
    int   confB = na
    bool bearC  = close < open
    bool bullC  = close > open
    float _pbBd = math.abs(close - open)
    float _pbWk = math.max(high - math.max(open, close), math.min(open, close) - low)
    bool  _pbQ  = not (pbBodyOn or pbBodyRef) or _pbBd > _pbWk
    bool pb     = (t.isHigh ? bearC : bullC) and _pbQ
    float _pbRf = pbBodyRef ? (t.isHigh ? math.min(open, close) : math.max(open, close)) : (t.isHigh ? low : high)
    bool newExt = na(t.potential) or (t.isHigh ? high >= t.potential : low <= t.potential)
    if newExt
        t.potential    := t.isHigh ? high : low
        t.potentialBar := bar_index
        t.count        := pb ? 1 : 0
        t.refLevel     := pb ? _pbRf : na
        t.swept        := false
    else
        if not na(swpLvl) and (t.isHigh ? low < swpLvl : high > swpLvl)
            t.swept := true
        if pb
            if t.count == 0
                t.count    := 1
                t.refLevel := _pbRf
            else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel) and t.swept
                confP := t.potential
                confB := t.potentialBar
                t.potential    := t.isHigh ? high : low
                t.potentialBar := bar_index
                t.count        := 1
                t.refLevel     := _pbRf
                t.swept        := false
            else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
                t.count    := 2
                t.refLevel := _pbRf
            else if t.count == 2 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
                confP := t.potential
                confB := t.potentialBar
                t.potential    := t.isHigh ? high : low
                t.potentialBar := bar_index
                t.count        := 1
                t.refLevel     := _pbRf
                t.swept        := false
    [confP, confB]

f_retExtreme(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd > 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd - 1))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd - 1
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

f_retExtInc(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd >= 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

method rearm(SwingTracker t, float p, int b) =>
    t.potential    := p
    t.potentialBar := b
    int  sz  = array.size(bufHi)
    int  idx = b - bar_index + sz - 1
    bool ok  = not na(b) and idx >= 0 and idx <= sz - 1
    int  dir = ok ? array.get(bufDir, idx) : 0
    bool pb  = t.isHigh ? dir == -1 : dir == 1
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? array.get(bufLo, idx) : array.get(bufHi, idx)) : na
    t.swept    := false

method armHere(SwingTracker t) =>
    t.potential    := t.isHigh ? high : low
    t.potentialBar := bar_index
    bool pb = t.isHigh ? close < open : close > open
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? low : high) : na
    t.swept    := false

f_isSwing(int pb, bool isHigh, int k) =>
    int  sz  = array.size(bufHi)
    int  idx = pb - bar_index + sz - 1
    bool ok  = not na(pb) and idx - k >= 0 and idx <= sz - 1
    if ok
        float p = isHigh ? array.get(bufHi, idx) : array.get(bufLo, idx)
        for j = 1 to k
            float v = isHigh ? array.get(bufHi, idx - j) : array.get(bufLo, idx - j)
            if isHigh ? v > p : v < p
                ok := false
    ok

f_confExt(array<int> arrB, array<float> arrP, bool wantLow, int afterBar, float beyond) =>
    float p = na
    int   b = na
    int   n = array.size(arrB)
    if n > 0

            int cb = array.get(arrB, i)
            if cb <= afterBar
                break
            float cp = array.get(arrP, i)
            if (wantLow ? cp <= beyond : cp >= beyond) and (na(p) or (wantLow ? cp < p : cp > p))
                p := cp
                b := cb
    [p, b]

            if cb == swpB

                break
            if cb < swpB
                break
    ok

f_hLine(line ln, label lb, bool on, float p, int b, string txt, color lc, color tc, string ls, int w, string ts, bool showT, string tv, string th) =>
    line  _l = ln
    label _b = lb
    if not on or na(p) or na(b)
        if not na(_l)
            line.delete(_l)
            _l := na
        if not na(_b)
            label.delete(_b)
            _b := na
    else
        int _x1 = b
        int _hMin = math.max(last_bar_index - histBars, bar_index - _xBack)
        if _x1 < _hMin
            _x1 := _hMin
        if _x1 < 0
            _x1 := 0
        if _x1 > bar_index
            _x1 := bar_index
        if na(_l)
            _l := line.new(_x1, p, bar_index, p, color = lc, width = w, style = lstyle(ls), extend = extend.right)
        else
            line.set_xy1(_l, _x1, p)
            line.set_xy2(_l, bar_index, p)
            line.set_color(_l, lc)
            line.set_width(_l, w)
            line.set_style(_l, lstyle(ls))
        if showT
            int _lx = th == "Left" ? _x1 : th == "Right" ? bar_index : int(math.avg(_x1, bar_index))
            if na(_b)
                _b := label.new(_lx, p, txt, color = color(na), textcolor = tc, style = lblStyle(tv), size = tsize(ts))
            else
                label.set_xy(_b, _lx, p)
                label.set_text(_b, txt)
                label.set_textcolor(_b, tc)
                label.set_style(_b, lblStyle(tv))
                label.set_size(_b, tsize(ts))
        else
            if not na(_b)
                label.delete(_b)
                _b := na
    [_l, _b]

var int stCntFill = 0
var int stCntCanc = 0
var int stCntTgt  = 0
var int stCntStop = 0
var int stCntOpp  = 0
var int stCntFlat = 0
var int stCntCap  = 0
var bool stCapFlag = false
var int stClosed  = 0
var int stWhy     = 0

f_stQty(float ent, float rsk) =>
    float pv = syminfo.pointvalue <= 0 ? 1.0 : syminfo.pointvalue
    float q = 0.0
    if stSize == "Fixed quantity"
        q := stQty
    else if stSize == "Fixed cash per trade"
        q := ent > 0 ? stCash / ent : 0.0
    else
        q := rsk > 0 ? stRisk / (rsk * pv) : 0.0
    float _lim = ent > 0 and stLev > 0 ? strategy.equity * stLev / ent : 0.0
    bool  _cap = _lim > 0 and q > _lim
    [_cap ? _lim : q, _cap]

if strategy.position_size != 0 and strategy.position_size[1] == 0
    stOpenSl := stSl
    stDir    := 0
    stCntFill += 1

if stDir != 0 and strategy.position_size == 0
    float _live = na
    if stDir == 1 and not na(florS)
        _live := florS.price
    if stDir == -1 and not na(ceilS)
        _live := ceilS.price
    if not na(_live)
        stSl := stDir == 1 ? _live - stSlBuf : _live + stSlBuf

if stFlatNow
    if strategy.position_size != 0
        strategy.close_all(comment = "Time flat")
        stWhy := 2
    if stDir != 0
        stCntCanc += 1
    strategy.cancel("L")
    strategy.cancel("S")
    stDir := 0

if not stWin
    strategy.cancel("L")
    strategy.cancel("S")
    if strategy.position_size == 0
        if stDir != 0
            stCntCanc += 1
        stDir := 0

bool stSigUp = evChoUp or (stBosOn and evBosUp)
bool stSigDn = evChoDn or (stBosOn and evBosDn)

if (stSigUp or stSigDn) and stOn
    if evChoUp or evChoDn
        stCntCho += 1
    if stBosOn and (evBosUp or evBosDn)
        stCntBos += 1
    bool _opp = (strategy.position_size > 0 and stSigDn) or (strategy.position_size < 0 and stSigUp)
    if _opp and stRev
        strategy.close_all(comment = "Opposite signal")
        stWhy := 1
    bool _take = stWin and (strategy.position_size == 0 or (_opp and stRev))
    if _take
        int   _d   = stSigUp ? 1 : -1
        float _piv = stSigUp ? stCeilPrev : stFlorPrev
        float _opl = na
        if _d == 1 and not na(florS)
            _opl := florS.price
        if _d == -1 and not na(ceilS)
            _opl := ceilS.price
        bool _agree = (_d == 1 and stT15 == 1) or (_d == -1 and stT15 == -1)
        bool _use   = _agree ? (stR2On and not na(_piv)) : stR1On
        if _use and not na(_opl)
            if stDir != 0
                stCntCanc += 1
            strategy.cancel("L")
            strategy.cancel("S")
            stCntArm += 1
            stCapFlag := false
            stDir  := _d
            stRule := _agree ? 2 : 1
            stSl   := _d == 1 ? _opl - stSlBuf : _opl + stSlBuf
            stFix  := _agree ? _piv : na

float stEnt = na
float stTgt = na
if strategy.position_size != 0 and not na(stOpenSl)
    stEnt := strategy.position_avg_price
    float _r = math.abs(stEnt - stOpenSl)
    stTgt := strategy.position_size > 0 ? stEnt + _r * stRR : stEnt - _r * stRR
else if stDir != 0
    stEnt := stRule == 2 ? stFix : f_retLvl(trendDir, mAnch, mOrig, stPbPct)
    if not na(stEnt) and not na(stSl)
        float _r = math.abs(stEnt - stSl)
        stTgt := stDir == 1 ? stEnt + _r * stRR : stEnt - _r * stRR

if stOn and strategy.position_size == 0 and stDir != 0 and stWin and not na(stEnt) and not na(stTgt) and not na(stSl)
    bool _sane = stDir == 1 ? stEnt > stSl : stEnt < stSl
    float _r = math.abs(stEnt - stSl)
    [_q, _qCap] = f_stQty(stEnt, _r)
    if _qCap and not stCapFlag
        stCntCap  += 1
        stCapFlag := true
    if _sane and _q > 0 and _r > 0
        if stDir == 1
            strategy.entry("L", strategy.long, qty = _q, limit = stEnt)
            strategy.exit("L exit", "L", stop = stSl, limit = stTgt)
        else
            strategy.entry("S", strategy.short, qty = _q, limit = stEnt)
            strategy.exit("S exit", "S", stop = stSl, limit = stTgt)
    else
        strategy.cancel("L")
        strategy.cancel("S")
        stCntCanc += 1
        stDir := 0

if strategy.position_size > 0 and not na(stOpenSl) and not na(stTgt)
    strategy.exit("L exit", "L", stop = stOpenSl, limit = stTgt)
if strategy.position_size < 0 and not na(stOpenSl) and not na(stTgt)
    strategy.exit("S exit", "S", stop = stOpenSl, limit = stTgt)

if strategy.closedtrades > stClosed
    int _nCl = strategy.closedtrades - stClosed
    for _i = 0 to _nCl - 1
        if stWhy == 1
            stCntOpp += 1
        else if stWhy == 2
            stCntFlat += 1
        else if strategy.closedtrades.profit(strategy.closedtrades - 1 - _i) > 0
            stCntTgt += 1
        else
            stCntStop += 1
    stWhy    := 0
    stClosed := strategy.closedtrades

stCeilPrev := not na(ceilS) ? ceilS.price : stCeilPrev
stFlorPrev := not na(florS) ? florS.price : stFlorPrev

plot(stShow and stDir != 0 and strategy.position_size == 0 ? stEnt : na, "Pending entry", color = color.new(color.orange, 0), style = plot.style_linebr, linewidth = 1)
plot(stShow and (stDir != 0 or strategy.position_size != 0) ? (strategy.position_size != 0 ? stOpenSl : stSl) : na, "Stop", color = color.new(color.red, 0), style = plot.style_linebr, linewidth = 1)
plot(stShow and (stDir != 0 or strategy.position_size != 0) ? stTgt : na, "Target", color = color.new(color.teal, 0), style = plot.style_linebr, linewidth = 1)
plotshape(stShow and stDir != 0 and stDir != nz(stDir[1], 0) and stRule == 2, "Rule 2 signal", shape.triangleup, location.belowbar, color.new(color.blue, 0), size = size.tiny)
plotshape(stShow and stDir != 0 and stDir != nz(stDir[1], 0) and stRule == 1, "Rule 1 signal", shape.triangledown, location.abovebar, color.new(color.purple, 0), size = size.tiny)

f_stPos(string p) =>
    p == "Top left" ? position.top_left : p == "Top center" ? position.top_center : p == "Top right" ? position.top_right : p == "Middle left" ? position.middle_left : p == "Middle right" ? position.middle_right : p == "Bottom left" ? position.bottom_left : p == "Bottom center" ? position.bottom_center : position.bottom_right

f_stCell(table t, int r, string a, string b, color c) =>
    table.cell(t, 0, r, a, text_color = color.silver, text_size = size.small, text_halign = text.align_left)
    table.cell(t, 1, r, b, text_color = c, text_size = size.small, text_halign = text.align_right)

var table stTbl = na
if stStatOn and bar_index == last_bar_index
    if na(stTbl)
        stTbl := table.new(f_stPos(stStatPos), 2, 11, bgcolor = color.new(color.black, 15), border_color = color.new(color.gray, 60), border_width = 1)
    table.cell(stTbl, 0, 0, "STRATEGY", text_color = color.white, text_size = size.small, bgcolor = color.new(color.blue, 55), text_halign = text.align_left)
    table.cell(stTbl, 1, 0, "count",    text_color = color.white, text_size = size.small, bgcolor = color.new(color.blue, 55), text_halign = text.align_right)
    f_stCell(stTbl, 1, "CHOCH signals",       str.tostring(stCntCho),  color.white)
    f_stCell(stTbl, 2, "BOS signals",         stBosOn ? str.tostring(stCntBos) : "off", stBosOn ? color.white : color.gray)
    f_stCell(stTbl, 3, "Setups armed",        str.tostring(stCntArm),  color.yellow)
    f_stCell(stTbl, 4, "Orders FILLED",       str.tostring(stCntFill), color.lime)
    f_stCell(stTbl, 5, "Armed, never filled", str.tostring(stCntCanc), color.orange)
    f_stCell(stTbl, 6, "Exit - target",       str.tostring(stCntTgt),  color.teal)
    f_stCell(stTbl, 7, "Exit - stop",         str.tostring(stCntStop), color.red)
    f_stCell(stTbl, 8, "Exit - opposite sig", str.tostring(stCntOpp),  color.fuchsia)
    f_stCell(stTbl, 9, "Exit - time flat",    str.tostring(stCntFlat), color.silver)
    f_stCell(stTbl, 10, "Size capped (leverage)", str.tostring(stCntCap), stCntCap > 0 ? color.orange : color.gray)
