f_addLn(array<line> la, array<int> cnt, line l) =>
    array.push(la, l)
    array.push(cnt, 0)

f_addLb(array<label> lb, array<int> cnt, label l) =>
    if array.size(cnt) > 0
        array.push(lb, l)
        int k = array.size(cnt) - 1
        array.set(cnt, k, array.get(cnt, k) + 1)

f_cutOld(array<line> la, array<label> lb, array<int> cnt, int cap) =>
    while array.size(la) > cap
        line.delete(array.shift(la))
        int c = array.shift(cnt)
        while c > 0 and array.size(lb) > 0
            label.delete(array.shift(lb))
            c -= 1

f_cutAll(array<line> la, array<label> lb, array<int> cnt) =>
    while array.size(la) > 0
        line.delete(array.pop(la))
    while array.size(lb) > 0
        label.delete(array.pop(lb))
    array.clear(cnt)

method repos(SLine s, int endBar) =>
    if not na(s) and not na(s.ln) and not na(s.lb)
        string th = hAlignOf(s.sid, not s.done)
        if th != "Left"
            s.lb.set_x(f_xClamp(th == "Right" ? endBar : int(math.avg(s.x1, endBar))))

method freeze(SLine s) =>
    if not na(s) and not na(s.ln)
        s.done := true
        [lc, tc, ls, w, txt, ts, tv, th, showT] = styleOf(s.sid, false)
        s.ln.set_x2(bar_index)
        s.ln.set_extend(extend.none)
        s.ln.set_color(lc)
        s.ln.set_width(w)
        s.ln.set_style(lstyle(ls))
        if not na(s.lb)
            s.lb.delete()
            s.lb := na
        if showT
            int lx = f_xClamp(th == "Left" ? s.x1 : th == "Right" ? bar_index : int(math.avg(s.x1, bar_index)))
            s.lb := label.new(lx, s.price, txt, color = color(na), textcolor = tc, style = lblStyle(tv), size = tsize(ts))
        f_addLn(mainLn, mainCnt, s.ln)
        if not na(s.lb)
            f_addLb(mainLb, mainCnt, s.lb)

method drop(SLine s) =>
    if not na(s) and not na(s.ln)
        s.ln.delete()
        if not na(s.lb)
            s.lb.delete()

type SwingTracker
    bool  isHigh
    float potential
    int   potentialBar
    int   count
    float refLevel
    bool  swept

method update(SwingTracker t, float swpLvl) =>
    float confP = na
    int   confB = na
    bool bearC  = close < open
    bool bullC  = close > open
    float _pbBd = math.abs(close - open)
    float _pbWk = math.max(high - math.max(open, close), math.min(open, close) - low)
    bool  _pbQ  = not (pbBodyOn or pbBodyRef) or _pbBd > _pbWk
    bool pb     = (t.isHigh ? bearC : bullC) and _pbQ
    float _pbRf = pbBodyRef ? (t.isHigh ? math.min(open, close) : math.max(open, close)) : (t.isHigh ? low : high)
    bool newExt = na(t.potential) or (t.isHigh ? high >= t.potential : low <= t.potential)
    if newExt
        t.potential    := t.isHigh ? high : low
        t.potentialBar := bar_index
        t.count        := pb ? 1 : 0
        t.refLevel     := pb ? _pbRf : na
        t.swept        := false
    else
        if not na(swpLvl) and (t.isHigh ? low < swpLvl : high > swpLvl)
            t.swept := true
        if pb
            if t.count == 0
                t.count    := 1
                t.refLevel := _pbRf
            else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel) and t.swept
                confP := t.potential
                confB := t.potentialBar
                t.potential    := t.isHigh ? high : low
                t.potentialBar := bar_index
                t.count        := 1
                t.refLevel     := _pbRf
                t.swept        := false
            else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
                t.count    := 2
                t.refLevel := _pbRf
            else if t.count == 2 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
                confP := t.potential
                confB := t.potentialBar
                t.potential    := t.isHigh ? high : low
                t.potentialBar := bar_index
                t.count        := 1
                t.refLevel     := _pbRf
                t.swept        := false
    [confP, confB]

var array<float> bufHi = array.new_float()
var array<float> bufLo = array.new_float()
var array<int>   bufDir = array.new_int()

f_retExtreme(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd > 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd - 1))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd - 1
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

f_retExtInc(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd >= 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

method rearm(SwingTracker t, float p, int b) =>
    t.potential    := p
    t.potentialBar := b
    int  sz  = array.size(bufHi)
    int  idx = b - bar_index + sz - 1
    bool ok  = not na(b) and idx >= 0 and idx <= sz - 1
    int  dir = ok ? array.get(bufDir, idx) : 0
    bool pb  = t.isHigh ? dir == -1 : dir == 1
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? array.get(bufLo, idx) : array.get(bufHi, idx)) : na
    t.swept    := false

method armHere(SwingTracker t) =>
    t.potential    := t.isHigh ? high : low
    t.potentialBar := bar_index
    bool pb = t.isHigh ? close < open : close > open
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? low : high) : na
    t.swept    := false

f_isSwing(int pb, bool isHigh, int k) =>
    int  sz  = array.size(bufHi)
    int  idx = pb - bar_index + sz - 1
    bool ok  = not na(pb) and idx - k >= 0 and idx <= sz - 1
    if ok
        float p = isHigh ? array.get(bufHi, idx) : array.get(bufLo, idx)
        for j = 1 to k
            float v = isHigh ? array.get(bufHi, idx - j) : array.get(bufLo, idx - j)
            if isHigh ? v > p : v < p
                ok := false
    ok

f_confExt(array<int> arrB, array<float> arrP, bool wantLow, int afterBar, float beyond) =>
    float p = na
    int   b = na
    int   n = array.size(arrB)
    if n > 0
        for i = n - 1 to 0
            int cb = array.get(arrB, i)
            if cb <= afterBar
                break
            float cp = array.get(arrP, i)
            if (wantLow ? cp <= beyond : cp >= beyond) and (na(p) or (wantLow ? cp < p : cp > p))
                p := cp
                b := cb
    [p, b]

f_swpConf(array<int> arrB, int swpB) =>
    bool ok = false
    int  n  = array.size(arrB)
    if not na(swpB) and n > 0
        for i = n - 1 to 0
            int cb = array.get(arrB, i)
            if cb == swpB
                ok := true
                break
            if cb < swpB
                break
    ok

f_lastMinor(bool wantLow, int beforeBar) =>
    arrB = wantLow ? msLoB : msHiB

    if n > 0
        for i = n - 1 to 0
            if array.get(arrB, i) < beforeBar
                p := array.get(arrP, i)
                b := array.get(arrB, i)
                break
    [p, b]

method armIdm(SLine s) =>
    if not na(s)
        bool isCeil = s.sid == 0 or s.sid == 2
        [ip, ib] = f_lastMinor(isCeil, s.x1)
        s.idmP   := ip
        s.idmB   := ib
        s.idmHit := false
        if not na(ip)

                if array.size(ab) == 2

    L - (f_dow(y, m, L) - w + 7) % 7
// a fixed-date holiday on a Saturday is kept on the Friday before, on a Sunday on the Monday after
f_obs(int y, int M, int D, int m, int d) =>
    int w = f_dow(y, M, D)
    m == M and d == (w == 7 ? D - 1 : w == 1 ? D + 1 : D)
// US market holiday on a New York date, "" if none. gf: count Good Friday (markets shut, government offices open)
f_usHol(int y, int m, int d, bool gf) =>
    string r = ""
    if m == 1 and ((d == 1 and f_dow(y, 1, 1) != 1 and f_dow(y, 1, 1) != 7) or (d == 2 and f_dow(y, 1, 1) == 1))
        r := "New Year"
    else if m == 1 and d == f_nthDow(y, 1, dayofweek.monday, 3)
        r := "Martin Luther King Day"
    else if m == 2 and d == f_nthDow(y, 2, dayofweek.monday, 3)
        r := "Presidents' Day"
    else if m == 5 and d == f_lastDow(y, 5, dayofweek.monday)
        r := "Memorial Day"
    else if y >= 2022 and f_obs(y, 6, 19, m, d)
        r := "Juneteenth"
    else if f_obs(y, 7, 4, m, d)
        r := "Independence Day"
    else if m == 9 and d == f_nthDow(y, 9, dayofweek.monday, 1)
        r := "Labor Day"
    else if m == 11 and d == f_nthDow(y, 11, dayofweek.thursday, 4)
        r := "Thanksgiving"
    else if f_obs(y, 12, 25, m, d)
        r := "Christmas"
    else if gf and (m == 3 or m == 4)
        // Easter Sunday (Gregorian computus); Good Friday is two days before
        int a  = y % 19
        int b  = int(y / 100)
        int c  = y % 100
        int h  = (19 * a + b - int(b / 4) - int((b - int((b + 8) / 25) + 1) / 3) + 15) % 30
        int l  = (32 + 2 * (b % 4) + 2 * int(c / 4) - h - c % 4) % 7
        int q  = int((a + 11 * h + 22 * l) / 451)
        [gy, gm, gd] = f_addD(y, int((h + l - 7 * q + 114) / 31), (h + l - 7 * q + 114) % 31 + 1, -2)
        if m == gm and d == gd
            r := "Good Friday"
    r
// NFP: the BLS rule - third Friday after the week (Sun-Sat) holding the 12th of the previous month
f_nfp(int y, int m) =>
    int py = m == 1 ? y - 1 : y
    int pm = m == 1 ? 12 : m - 1
    [y1, m1, d1] = f_addD(py, pm, 12 + (7 - f_dow(py, pm, 12)) % 7, 20)
    int ry = y1
    int rm = m1
    int rd = d1
    if rm == 1 and rd <= 3
        [y2, m2, d2] = f_addD(ry, rm, rd, 7)
        ry := y2
        rm := m2
        rd := d2
    if f_usHol(ry, rm, rd, false) != ""
        [y3, m3, d3] = f_addD(ry, rm, rd, -1)
        ry := y3
        rm := m3
        rd := d3
    [ry, rm, rd]
// n-th US working day of the month this date is (0 = not a working day, or later than the 10th)
f_bday(int y, int m, int d) =>
    int n = 0
    if d <= 10 and f_dow(y, m, d) != 1 and f_dow(y, m, d) != 7 and f_usHol(y, m, d, false) == ""
        for k = 1 to d
            int w = f_dow(y, m, k)
            if w != 1 and w != 7 and f_usHol(y, m, k, false) == ""
                n += 1
    n

// v7.4 STOP MANAGEMENT (group 32): one record per open trade, only while a switch is ON

f_stTrk(string id) =>
    StTrade _r = na
    int _n = array.size(stTrk)
    if _n > 0
        for _k = 0 to _n - 1
            StTrade _t = array.get(stTrk, _k)

if _eNew
    stOpenSl   := na(stLastSl)  ? stSl      : stLastSl
    stOpenTgt  := na(stLastTgt) ? stPlanTgt : stLastTgt
    // v8.2: one record per open trade - its stop and target, for the safety net and for group 32
    if _eOpen and not na(stOpenSl) and not na(stOpenTgt)

        for _k = strategy.opentrades - 1 to 0
            if strategy.opentrades.entry_bar_index(_k) != stEntNow
                break

                int   _kD  = strategy.opentrades.size(_k) > 0 ? 1 : -1
                bool  _kPp = stPpSent and _kId == stPendId + "p"
                float _r0 = math.abs(_kP - stOpenSl)
                // break-even = entry + the round-turn costs of this size, so a break-even exit nets zero
                float _cst = _stPvA > 0 ? (stCmU + (_kQ > 0 ? f_stCmF() / _kQ : 0.0)) / _stPvA : 0.0
                array.push(stTrk, StTrade.new(_kId, _kD, _kP, _r0, _kD == 1 ? _kP + _cst : _kP - _cst, stOpenSl, _kPp and not na(stLastTp1) ? stLastTp1 : stOpenTgt, _kQ, 0, bar_index, false, -1, "", _kPp ? 1 : 0, false))
    // v9.0 e: the two parts of a setup fill together; a part still waiting is cancelled, so it can never fill on its own later

            strategy.cancel(stPendId + " exit")
            stPpCnl := true
        if not f_stFilled(stPendId + "p")
            strategy.cancel(stPendId + "p")
            strategy.cancel(stPendId + "p exit")
            stPpCnl := true
    stPpSent := false
    if stRefId != "" and stPendId == stRefId
        stCntRef -= 1
        stRefId  := ""
    if stDir != 0
        f_audEnd("FILLED", stAudFillC)
    stDir      := 0
    stPendId   := ""
    stOrdLive  := false
    stCntFill   += 1
    stDayTrades += 1
    if stRule == 3
        stCntR3 += 1
    else if stRule == 2
        stCntR2 += 1
    else

    if stLastCapQ

    stLastRisk := strategy.convert_to_account(_eQ * math.abs(_eP - stSl) * f_stPv()) + _eQ * stCmU + f_stCmF()
    stCmLast   := _eQ * stCmU + f_stCmF()
    stLastQty  := _eQ
    stEntSeen  := stEntNow

// yet TradingView did not fill it. TradingView refuses an order the account cannot hold (Properties leverage /
// margin too low for the size) and one above Properties pyramiding. A refused order sends no alert, so the
// broker gets no trade either. Counted once per setup, taken off again if it fills later. It changes no trade.
if not _eNew and stOrdLive and stDir != 0 and stPendId != "" and stPendId != stRefId and not na(stLastEnt) and (stDir == 1 ? low < stLastEnt : high > stLastEnt)
    // v8.1: an order that DID fill (only the fill check missed it) is not refused - look for it among the trades
    bool _filled = f_stIsOpen(stPendId)
    if not _filled and strategy.closedtrades > 0
        for _i = strategy.closedtrades - 1 to math.max(0, strategy.closedtrades - 50)
            if strategy.closedtrades.entry_id(_i) == stPendId
                _filled := true
                break
    if not _filled
        stCntRef += 1
        stRefId  := stPendId

        if str.endswith(_xid, " exit")

            float   _xp  = strategy.closedtrades.exit_price(_ix)

                stCntMvx += 1

            else

        else if _part

        else if stWhy == 2

        else if stWhy == 3

        else if stWhy == 4

            else
                stCntStop += 1
        else

        stDayPnl  := stDayPnl + _pf

        if not (str.endswith(_gId, "p") and map.contains(stCodeOf, _gId))
            float _gR = math.abs(nz(strategy.closedtrades.commission(_i)))
            float _gE = math.abs(nz(strategy.closedtrades.size(_i))) * f_stCmU(nz(strategy.closedtrades.entry_price(_i), close)) + f_stCmF()
            float _gP = strategy.closedtrades.profit(_i) - (_gR > 0 ? 0.0 : _gE)

                    stCntLd   += 1

// v8.2 SAFETY NET, every candle, for every open trade. After its entry candle, if the price went more than 2 ticks
// beyond its stop or its target and the trade is still open, TradingView missed its bracket; and if the strategy asked
// for all trades to close on an earlier candle and this one is still open, TradingView missed the close. Either way it
// is counted (table row, red) and the strategy closes the trade itself at the next open, with its own close message.
// v8.3: the net's close is sent at the END of the candle (see "the candle's last orders"). If it is still not done a
// candle later, the net asks again for this trade only, every candle - v8.2 used strategy.close_all here, which closed
// every trade in TradingView but sent only one trade's tag to your broker.
// v7.4: then, at the candle close, move the stop of every managed open trade that reached its next step. The bracket
// is sent again with the new stop - same id, same target, same close message - so from the next tick the moved stop
// is hit in real time like any stop. A trade that has ended is dropped from the list.
if array.size(stTrk) > 0
    for _i = array.size(stTrk) - 1 to 0

            bool  _pastTg = bar_index > _t.bar and (_t.dir == 1 ? high > _t.tgt + _tk2 : low < _t.tgt - _tk2)
            bool  _noCls  = stCloseReq >= 0 and bar_index > stCloseReq and _t.bar <= stCloseReq

                    else if _t.step == 0 and (_t.dir == 1 ? close >= _t.ent + _t.r : close <= _t.ent - _t.r)
                        _k := 1
                    // the new stop must lie below the level just reached (above it for a sell) - with costs bigger than 1R,
                    // break-even is above 1R and the move waits for a later step
                    float _c = _k == 1 ? _t.be : (_t.dir == 1 ? math.max(_t.be, _t.ent + (_k - 1) * _t.r) : math.min(_t.be, _t.ent - (_k - 1) * _t.r))
                    if _k > _t.step and (_t.dir == 1 ? _c < _t.ent + _k * _t.r : _c > _t.ent - _k * _t.r)
                        _ns := _t.dir == 1 ? math.max(_ns, _c) : math.min(_ns, _c)
                        _t.step := _k
                // v9.0 c: structure trailing stop - behind the CHOCH* level or behind every new swing, never backwards
                if stTrOn and bar_index > _t.bar
                    float _lv = na
                    if stTrMode == "Behind every new swing (faster)"
                        _lv := _t.dir == 1 ? stSwL : stSwH
                    else if _t.dir == 1 and trendDir == 1 and not na(florS)
                        _lv := florS.price
                    else if _t.dir == -1 and trendDir == -1 and not na(ceilS)
                        _lv := ceilS.price
                    if not na(_lv)

                        if _t.dir == 1 ? _l2 < close : _l2 > close
                            _ns := _t.dir == 1 ? math.max(_ns, _l2) : math.min(_ns, _l2)
                // v9.0 e: the early-profit part of this setup hit its target - the rest goes to break-even
                if _t.pp and stPpBe
                    _ns := _t.dir == 1 ? math.max(_ns, _t.be) : math.min(_ns, _t.be)
                if _ns != _t.sl
                    _t.sl := _ns
                    _t.mv := true
                    strategy.exit(_t.id + " exit", _t.id, stop = _ns, limit = _t.tgt, alert_message = f_stMsg("close", _t.dir, close, _ns, _t.tgt, _t.q, _t.id))
                    stCntMoves += 1
                    if _t.id == strategy.opentrades.entry_id(strategy.opentrades - 1)
                        stOpenSl := _ns
