//  Built from Market_Structure_SMC v15.9 by removing every volume-profile and
//  VWAP module and adding an EMA module. EVERY STRUCTURE RULE IS UNCHANGED: the
//  engine block is byte-identical to v15.9.
//
//  v1.10  THE MTF TABLE STOPS AT THE HIGHEST TIMEFRAME THE CHART CAN
//  ACTUALLY FEED IT.  (v1.8 is withdrawn; this is built on v1.7.)
//    THE PROBLEM, measured rather than assumed. A probe requesting
//    bar_index + 1 from each timeframe reported how many bars request.security
//    really returns. On EURUSD, from a 1m chart, the weekly series came back
//    with NINE bars and the monthly with ONE. The per-timeframe engine is a
//    state machine that replays every pivot and break from the first bar it can
//    see, so one monthly candle does not make it fail — it makes it answer
//    confidently and wrongly. That is where the monthly row reading BULLISH on
//    a 15m chart and BEARISH on a 1h chart came from.
//    THE RULE, which fixes it by not asking the question. A row is drawn only
//    when its timeframe is at or above the chart AND at or below the ceiling
//    the chart can support:
//        chart 5m and below ....... ceiling 4h   (nothing daily or higher)
//        chart 15m ................ ceiling 1W   (daily and weekly, no monthly)
//        chart 10m, 30m, 45m, 1h .. ceiling 1D   (daily is the top row)
//        chart 2h and above ....... ceiling 1M   (everything, through monthly)
//    A 2m chart falls in with 5m and below, and the 61-to-119-minute band with
//    1h and below; neither is named by the rule and both take the conservative
//    side. THE 5m-AND-BELOW CEILING IS WHY: from a 1m chart the daily series
//    comes back with about SIXTY-NINE bars, and on the reported EURUSD chart
//    that row read BULLISH while the daily chart itself read BEARISH. Sixty-
//    nine bars is not a daily structure, so the row is no longer offered.
//    IT IS ALSO THE SECURITY-CALL OPTIMISATION. A row outside the ceiling has
//    its _uX flag false, and a false flag already collapsed that request to
//    timeframe.period — the chart\'s own data, already loaded, costing nothing.
//    So on a 1m chart the weekly and monthly datasets are no longer fetched at
//    all. Fewer real datasets per script is exactly what keeps an alert inside
//    the server\'s budget, which is where the calculation errors came from.
//    The requests stay DYNAMIC, as asked. The same mechanism means an alert
//    created with the table switched off fetches no extra timeframe at all,
//    because tblOn is part of every flag.
//    TIDIER AND CHEAPER: each of the eighteen flag lines used to call
//    timeframe.in_seconds on its own string, and now calls it once into a
//    named constant that both tests read. The flags stay plain inline
//    expressions on purpose — wrapping them in a helper function would let
//    Pine promote them from simple to series, and a series flag would make
//    the timeframe argument of request.security series too, which is exactly
//    the thing being avoided here.
//    AND ONE MICRO-OPTIMISATION in the table renderer: tsize(tblRowSz) was
//    being recomputed six times per row on every redraw. It is hoisted to a
//    single value.
//    NO INPUT is added, removed or moved, so saved settings carry over and no
//    settings reset is needed. NOT ONE ENGINE RULE CHANGES: the chart
//    drawings, the three tiers, the alerts and every structure classification
//    are identical to v1.7.
//
//  v1.7  ONE CANDLE CAN NO LONGER SUPPLY A LEVEL AND TRIGGER ITS OWN
//  PROMOTION ON THE SAME BAR.
//    Reported on two charts. FIRST: a single wide candle swept the ceiling AND
//    the floor, and both pivot-2 promotions fired on that one bar, so the whole
//    range collapsed onto that candle — its high became the possible BOS and its
//    low the possible CHOCH, neither of them a pivot anything had confirmed.
//    SECOND: a BOS line anchored back at its original pivot but drawn at the low
//    of a much later candle, because that candle wicked below the floor, closed
//    back above it, and its own high cleared the opposite reference — sweep and
//    trigger on one bar again, so the floor was promoted down to the wick of the
//    very candle that made it.
//    THE CAUSE, and it is the same one both times. The promotion asks two
//    questions: was the level swept, and has price since taken out the opposite
//    reference. Nothing stops BOTH from being answered by the SAME candle. A
//    wide bar wicks through the level and, in the same range, trades past the
//    reference on the other side — so the sweep is recorded and the trigger
//    fires before the bar has even closed, and the level moves to a wick that
//    is one bar old and has been confirmed by nothing.
//    THE FIX, and it is deliberately the narrowest one that works. When the
//    trigger is satisfied ON THE BAR THAT MADE THE SWEPT EXTREME, that sweep is
//    flagged, and while it carries the flag the promotion additionally requires
//    the swept wick to have been confirmed as a pivot by the 3-CANDLE PULLBACK
//    RULE. A wick made on the current bar can never be confirmed on its own bar,
//    so the collapse cannot happen; if the pullback arrives later, the promotion
//    fires then, exactly as before. THE FLAG IS PER SWEEP: recording a new,
//    deeper swept extreme clears it, and a level re-created by any route starts
//    clean.
//    WHAT IS NOT TOUCHED, on purpose. If the trigger arrives on a LATER bar than
//    the sweep — which is the ordinary case, and the GBPCHF sequence v15.14
//    change 2 was written for — nothing is flagged, nothing is checked, and the
//    promotion behaves exactly as it did in v1.6. v15.14 change 2 is intact:
//    an unconfirmed swept wick still takes the level. Only the one-candle case
//    is gated.
//    WHAT THE FIRST CHART NOW DOES: the ceiling is not promoted, so it stays on
//    the original pivot and the break is measured there. The candle then closes
//    through it and the opposite side re-anchors with f_retExtInc, whose window
//    includes the break candle (v15.14 change 1) — so the new floor lands on
//    that candle's low, the deepest point of the leg. The original pivot keeps
//    the BOS, the newest low becomes the CHOCH*, which is the outcome asked for.
//    ALL SIX PROMOTION SITES: main, internal and inner, up-break and down-break.
//    ONE FIELD is added to the SLine type and one bool to each of the four
//    internal and inner sweep records. NO INPUT is added, removed or moved, so
//    saved settings carry over and no settings reset is needed. The 3-candle
//    rule itself, the swing detector, the IDM gate, the alternation, the
//    retracement alerts and every drawing rule are untouched.
//
//  v1.6  INNER ALERTS COMPLETED · TWO NEW LIVE-LINE PAIRS · 50% RANGE LINES ·
//  A MANUAL LIMIT ON HOW MANY MARKS STAY ON THE CHART.
//  Four additions, all requested, and every one of them appended at the end of
//  the script so that no existing input moves. Saved settings carry over and no
//  settings reset is needed. Not one engine rule is touched: the classification
//  of every BOS, CHOCH, iBOS, iCHOCH, nBOS and nCHOCH is identical to v1.5,
//  and the diff to the engine block is four lines, all of them pushing a
//  drawing that was already being created into an array so it can be counted.
//
//  1. THE INNER TIER CAN NOW BE ALERTED ON FROM THE CONDITION DROPDOWN.
//     The inner engine has fired alert() calls for bullish nBOS, bearish nBOS,
//     bullish nCHOCH, bearish nCHOCH and "any inner structure event" since the
//     tier was added, but NOT ONE of them had an alertcondition entry — and
//     alertcondition is what puts an event in the Condition dropdown of
//     TradingView's Create Alert dialog. The main and internal tiers had their
//     entries; the inner tier was reachable only through "Any alert() function
//     call". SEVEN alertcondition entries are added: Bullish nBOS, Bearish
//     nBOS, Bullish nCHOCH, Bearish nCHOCH, Any nBOS, Any nCHOCH and Any inner
//     structure event. Two alert() toggles are added with them — "Any nBOS
//     alert" and "Any nCHOCH alert" — because those two combined streams
//     existed for the internal tier and not for the inner one. Every new entry
//     obeys the arming gate exactly like the ones already there.
//     The five arming slots also gain nBOS, nCHOCH and "Any inner structure
//     event" as triggers, so an inner event can now arm the gate as well as
//     pass through it.
//
//  2. iBOS* / iCHOCH* AND nBOS* / nCHOCH* LINES.
//     The main engine draws its two live levels as the extended BOS* and CHOCH*
//     lines. The internal and inner engines tracked exactly the same pair of
//     levels and drew nothing at all until one of them broke, so the level
//     price was about to test was invisible on two of the three tiers. Both
//     pairs are now drawn, on the main engine's own naming rule: the level
//     whose break would print a BOS gets the *BOS* style, the level whose break
//     would print a CHOCH gets the *CHOCH* style, and a line therefore changes
//     name and style the instant its tier's trend flips. While a tier's trend
//     is still neutral — which every fresh range starts out as — BOTH of its
//     lines read *CHOCH*, because the first break in either direction is a
//     CHOCH. Four new style groups, each with the full set the main groups
//     have: line colour, text colour, style, width, label text, text size,
//     show-text, vertical and horizontal placement. Two on/off switches.
//     They are redrawn in place on the last bar, so the pair costs four
//     drawings no matter how long the chart is, and they can never crowd out
//     the iBOS / iCHOCH history.
//
//  3. THE 50% LINE OF EVERY RANGE.
//     The equilibrium of the BOS*/CHOCH* pair, of the iBOS*/iCHOCH* pair and of
//     the nBOS*/nCHOCH* pair. Each has its own percentage — 50 by default, so
//     it is the midpoint — measured UP FROM THE LOWER of the two lines, which
//     is what makes 50 read the same whichever way the tier's trend is running.
//     Each has the same complete style set the structure groups have, its own
//     on/off switch, and an option to write the percentage into the label so it
//     cannot go stale when the number is changed. One shared setting decides
//     where all three lines start: the older level, the newer level, or the
//     current bar.
//     A LINE NEEDS BOTH OF ITS LEVELS. Every break empties the side it broke
//     and it stays empty until the 3-candle rule confirms the next pivot, so
//     there is no range to halve in between and the line is simply not drawn.
//     The main line is on by default, the internal and inner ones are off,
//     because three equilibrium lines at once is a busy chart.
//
//  4. MAX MARKS TO KEEP, ONE SETTING PER TIER.
//     "Max BOS / CHOCH marks to keep" sits in the Engine group, "Max iBOS /
//     iCHOCH" and "Max nBOS / nCHOCH" in the Internal + Inner group. 0 on all
//     three keeps the behaviour every earlier version had. Any other number
//     keeps that many of the most recent structure lines on that tier and
//     deletes the rest oldest-first, along with their labels and their strong-
//     point stamps. Set 3 and the chart carries the last three breaks only.
//     DISPLAY ONLY, all three: the engine still computes every break on every
//     bar, and every alert still fires whatever these are set to. The live
//     BOS* / CHOCH* pair and the two new live pairs are never counted or
//     deleted by them — those are the current levels, not history.
//
//  v1.5  THE PIVOT-2 PROMOTION NO LONGER MOVES THE ANCHOR.
//    Reported on BTCUSD 15m after v15.14: a main BOS and a main CHOCH printing
//    on adjacent candles a few points apart.
//    THE CAUSE, and it is v15.5 all over again. v15.14 promoted the level to
//    the swept wick AND set the new line x1 to the SWEEP BAR:
//        int sbC = ceilS.swpB
//        ceilS := mkSLine(spC, sbC, sdC, drawOk)
//    When that ceiling is later broken the opposite side re-anchors with
//    f_retExtInc(ceilS.x1, false), which scans from x1 + 1. With x1 sitting on
//    the sweep bar, the SWEEP CANDLE OWN LOW is outside the window — and so is
//    everything between the original pivot and the sweep. refP never covered
//    the sweep bar either, so nothing else compensates. The new floor comes out
//    shallower than the leg really was, and the next candle closes straight
//    back through it: BOS, then CHOCH, one bar apart.
//    v15.5 fixed the identical fault for the equal-price retest and stated it
//    plainly: move x1 forward and that window shrinks, so the floor lands on a
//    shallow low inside the real leg. Invariant 2 says the same thing —
//    re-creating the trend-side line inside a range must not re-scope it.
//    THE FIX: the promotion takes the swept wick PRICE and keeps the ORIGINAL
//    anchor bar, at all six sites. The rule you asked for in v15.14 is intact —
//    the level still becomes the swept wick — only the scope is preserved.
//    MEASURED over 83,239 randomised promotion-then-break sequences: the two
//    give a different floor in 9,035 of them (10.9%), and the kept anchor is
//    deeper or equal in every single one. It never raises a floor.
//    Six lines. No input added, removed or moved; no settings reset needed.
//
//  v1.4  TWO RULE CHANGES, BOTH REQUESTED, BOTH IN ALL THREE ENGINES.
//
//  1. THE BREAK CANDLE NOW COUNTS TOWARDS THE OPPOSITE LEVEL.
//     Reported on GBPCAD 4h: candles 1-4 formed the pullback that confirmed a
//     swing high; candle 5 wicked BELOW candle 4 low, then closed above the
//     ceiling and printed the iBOS. The new floor landed on candle 4 low, not
//     on candle 5 wick, even though candle 5 wick was the deepest point of that
//     pullback and happened before the break close.
//     THE CAUSE, one clause in f_retExtreme:
//         int jEnd = sz - 1          // index of the CURRENT bar
//         for j = j0 to jEnd - 1     // <- the break candle is skipped
//     It implemented the documented rule literally — "the extreme wick STRICTLY
//     BETWEEN the broken pivot candle and the break candle, both excluded".
//     Excluding the pivot candle is right. Excluding the BREAK candle is not:
//     it discards the deepest point of the leg in exactly the case that matters
//     most, where the candle sweeps the previous low, takes the liquidity, and
//     then closes up through the level. That wick is the strong low.
//     THE FIX: f_retExtInc, identical except that it scans to jEnd, is used at
//     the SIX opposite-side re-anchors — main, internal and inner, up-break and
//     down-break. It can only ever deepen the level, and only when the break
//     candle genuinely made the leg extreme; on an ordinary break candle the
//     two functions return the same bar, so nothing else moves.
//     NOT CHANGED, deliberately: the six opposite-REFERENCE calls that arm the
//     pivot-2 promotion, the two IDM taken-past tests, and the two Module 1
//     alternation re-arms all still use the exclusive f_retExtreme.
//     The strong-point stamps follow the new level, because they read the same
//     result — so the marks now sit on the sweep wick.
//
//  2. A PIVOT-2 PROMOTION TAKES THE SWEPT WICK.
//     Reported on GBPCHF 4h: P1 was the ceiling and P2 the floor, confirmed by
//     the P1-P2 pullback satisfying the 3-candle rule. P3 then wicked ABOVE P1
//     and closed back inside — a sweep — and P4 took out P2. P4 became the
//     possible CHOCH, but P3 never became the possible BOS, so the later break
//     was measured against P1 at 1.0922 instead of P3 at 1.0944.
//     THE CAUSE: the promotion called f_confExt, which requires the replacement
//     to be a 3-candle-CONFIRMED pivot (v13.4, extended to internal in v13.6).
//     P3 high was never confirmed, so f_confExt returned na, no promotion fired
//     and the original pivot kept the line — which is what the rulebook said.
//     THE RULE, as requested: once a pullback has satisfied the 3-candle rule,
//     no later pullback has to satisfy it again until a break happens. So the
//     swept extreme itself takes the level.
//     THE FIX: the six promotions now take the recorded sweep extreme
//     (ceilS.swpP / florS.swpP, iCeilSwp / iFlorSwp, nCeilSwp / nFlorSwp) and
//     its bar, instead of asking f_confExt for a confirmed pivot. f_confExt has
//     no callers left; the definition is kept so this is easy to revert.
//     BOTH TRIGGERS ARE UNCHANGED. A promotion still requires a genuine sweep
//     of the level AND price then taking out the opposite reference. This is
//     not "any wick can become a level".
//     IT DOES REVERSE INVARIANT 4 for this one path, which v13.4 added on
//     purpose. Invariant 4 has been reworded rather than quietly broken. If a
//     mark ever lands on a wick you do not consider structure, this is the
//     first thing to look at.
//
//  MEASURED IN NODE against both reported sequences: change 1 moves the floor
//  from candle 4 low to the break candle wick and is inert on an ordinary break
//  candle; change 2 leaves the ceiling on P1 under the old code and promotes it
//  to P3 swept wick under the new one, and is inert when no sweep occurred.
//  NO INPUT is added, removed or moved, so saved settings carry over and no
//  settings reset is needed. Module 1, the 3-candle rule itself, the alternating
//  and free-running streams, the retracement alerts and the arming gate are all
//  untouched. Existing alerts should be recreated only if you want them to see
//  the new levels.

//    NEW INPUT: "▶ EMA — show exponential moving averages", off by default,
//    drawn in group 0 Master Visibility Toggles alongside MASTER A and B.
//    While it is off the module is completely inert: no line, no ribbon, no
//    label, and no EMA cross alert can fire. Turn it on and the EMAs appear
//    exactly as group 9 has them set — 9, 21, 50 and 200, with 100 available.
//    IT IS A GATE, NOT AN OVERRIDE. Unlike MASTER A and MASTER B, which force

//  THE 3-CANDLE PULLBACK RULE (Module 1) — swing HIGH; lows mirror
//    1. CANDIDATE = highest high so far; a new OR EQUAL extreme replaces it.
//    2. Three BEARISH pullback candles: c1 = first bearish candle at/after the
//       candidate (its LOW locks); c2 must CLOSE below c1 low; c3 must CLOSE
//       below c2 low -> CONFIRMED.
//    3. Always body-close beyond the previous locked WICK. Never body-to-body.
//    4. Noise PAUSES, never resets. The three need not be consecutive.
//    5. A new OR EQUAL extreme RESTARTS everything.
//    6. Reported on the confirmation bar, ANCHORED at the candidate bar.
//
//  TWO PIVOT STREAMS — never mix them up
//    MAIN      hiT / loT   free-running.  Read by BOS / CHOCH only.
//    INTERNAL  aHiT / aLoT alternating.   Read by iBOS / iCHOCH and nBOS / nCHOCH.
//
//  THREE NESTED ENGINES
//    MAIN     BOS / CHOCH    · anchored on the absolute extreme at chart start
//    INTERNAL iBOS / iCHOCH  · anchored on the main trend-side line pivot
//    INNER    nBOS / nCHOCH  · anchored on the internal trend-side level
//  Each: adopt into an EMPTY slot — an equal-price pivot never takes the level
//  (v15.5) · first break of a range = CHOCH · then with-trend = BOS, against =
//  CHOCH + flip · broken side empties and refills · opposite side re-anchors to
//  the retracement extreme · sweeps arm a pivot-2 promotion.
//
//  INVARIANTS — break these and old bugs come back
//    1. Only a confirmed MAIN event starts a new internal range; only a confirmed
//       INTERNAL event starts a new inner range.
//    2. The range anchor is LATCHED for the life of the range.
//    3. Main = free-running pivots. Internal and inner = alternating.
//    4. A structure level is a CONFIRMED PIVOT, with ONE deliberate exception
//       (v15.14): a pivot-2 promotion takes the SWEPT WICK. There the sweep of
//       the level plus price then taking out the opposite reference IS the
//       qualification, so no separate 3-candle confirmation is required.
//    5. Never adopt or seed a level price has already closed beyond.
//    6. NEW INPUTS MUST BE APPENDED AT THE END — TradingView matches saved
//       settings by declaration ORDER. Use group = to place them on screen.
//    7. rearm() reads the chart price buffers, so it must never be called inside

//    8. No variable-offset history references — use bufHi / bufLo / bufDir.
//    9. Pine has no bitwise operators; the EVENT bitmask is unpacked by

//
//  ARMING GATE (v15.6): a price slot arms only when price REACHES the level from
//  the side it is actually on — close[1] fixes the side, then the wick must come

f_hhmm(int h, int m) =>
    (h < 10 ? "0" : "") + str.tostring(h, "0") + (m < 10 ? "0" : "") + str.tostring(m, "0")

f_sesShift(string ses, int hrs) =>
    string out = ses
    if hrs != 0
        int dash = str.pos(ses, "-")
        if not na(dash) and dash >= 4 and str.length(ses) >= dash + 5
            float ahf = str.tonumber(str.substring(ses, dash - 4, dash - 2))
            float amf = str.tonumber(str.substring(ses, dash - 2, dash))
            float bhf = str.tonumber(str.substring(ses, dash + 1, dash + 3))
            float bmf = str.tonumber(str.substring(ses, dash + 3, dash + 5))
            if not na(ahf) and not na(amf) and not na(bhf) and not na(bmf)
                int ah = (int(ahf) + hrs) % 24
                int bh = (int(bhf) + hrs) % 24
                if ah < 0
                    ah += 24
                if bh < 0
                    bh += 24
                string tail = str.length(ses) > dash + 5 ? str.substring(ses, dash + 5) : ""
                out := f_hhmm(ah, int(amf)) + "-" + f_hhmm(bh, int(bmf)) + tail
    out

hAlignOf(int sid, bool pend) =>
    if pend
        sid <= 1 ? p1_th : p2_th
    else
        switch sid
            0 => b1_th
            1 => b2_th
            2 => c1_th
            => c2_th

method repos(SLine s, int endBar) =>
    if not na(s) and not na(s.ln) and not na(s.lb)
        string th = hAlignOf(s.sid, not s.done)
        if th != "Left"
            s.lb.set_x(th == "Right" ? endBar : int(math.avg(s.x1, endBar)))

method freeze(SLine s) =>
    if not na(s) and not na(s.ln)
        s.done := true
        [lc, tc, ls, w, txt, ts, tv, th, showT] = styleOf(s.sid, false)
        s.ln.set_x2(bar_index)
        s.ln.set_extend(extend.none)
        s.ln.set_color(lc)
        s.ln.set_width(w)
        s.ln.set_style(lstyle(ls))
        if not na(s.lb)
            s.lb.delete()
            s.lb := na
        if showT
            int lx = th == "Left" ? s.x1 : th == "Right" ? bar_index : int(math.avg(s.x1, bar_index))
            s.lb := label.new(lx, s.price, txt, color = color(na), textcolor = tc, style = lblStyle(tv), size = tsize(ts))
        array.push(mainLn, s.ln)
        if not na(s.lb)
            array.push(mainLb, s.lb)

method drop(SLine s) =>
    if not na(s) and not na(s.ln)
        s.ln.delete()
        if not na(s.lb)
            s.lb.delete()

mkSLine(float p, int b, int sid, bool draw) =>
    [lc, tc, ls, w, txt, ts, tv, th, showT] = styleOf(sid, true)
    line ln = na

    if draw

        lb := label.new(lx, p, txt, color = color(na), textcolor = tc, style = lblStyle(tv), size = tsize(ts))
    SLine.new(ln, lb, b, p, sid, bar_index, false, na, na, na, na, na, false, false)

type SwingTracker
    bool  isHigh
    float potential
    int   potentialBar
    int   count
    float refLevel

method update(SwingTracker t) =>
    float confP = na
    int   confB = na
    bool bearC  = close < open
    bool bullC  = close > open
    bool pb     = t.isHigh ? bearC : bullC
    bool newExt = na(t.potential) or (t.isHigh ? high >= t.potential : low <= t.potential)
    if newExt
        t.potential    := t.isHigh ? high : low
        t.potentialBar := bar_index
        t.count        := pb ? 1 : 0
        t.refLevel     := pb ? (t.isHigh ? low : high) : na
    else if pb
        if t.count == 0
            t.count    := 1
            t.refLevel := t.isHigh ? low : high
        else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
            t.count    := 2
            t.refLevel := t.isHigh ? low : high
        else if t.count == 2 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
            confP := t.potential
            confB := t.potentialBar
            t.potential    := t.isHigh ? high : low
            t.potentialBar := bar_index
            t.count        := 1
            t.refLevel     := t.isHigh ? low : high
    [confP, confB]

f_retExtreme(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd > 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd - 1))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd - 1
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

f_retExtInc(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd >= 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

method rearm(SwingTracker t, float p, int b) =>
    t.potential    := p
    t.potentialBar := b
    int  sz  = array.size(bufHi)
    int  idx = b - bar_index + sz - 1
    bool ok  = not na(b) and idx >= 0 and idx <= sz - 1
    int  dir = ok ? array.get(bufDir, idx) : 0
    bool pb  = t.isHigh ? dir == -1 : dir == 1
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? array.get(bufLo, idx) : array.get(bufHi, idx)) : na

method armHere(SwingTracker t) =>
    t.potential    := t.isHigh ? high : low
    t.potentialBar := bar_index
    bool pb = t.isHigh ? close < open : close > open
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? low : high) : na

f_isSwing(int pb, bool isHigh, int k) =>
    int  sz  = array.size(bufHi)
    int  idx = pb - bar_index + sz - 1
    bool ok  = not na(pb) and idx - k >= 0 and idx <= sz - 1
    if ok
        float p = isHigh ? array.get(bufHi, idx) : array.get(bufLo, idx)
        for j = 1 to k
            float v = isHigh ? array.get(bufHi, idx - j) : array.get(bufLo, idx - j)
            if isHigh ? v > p : v < p
                ok := false
    ok

f_confExt(array<int> arrB, array<float> arrP, bool wantLow, int afterBar, float beyond) =>
    float p = na
    int   b = na
    int   n = array.size(arrB)
    if n > 0
        for i = n - 1 to 0
            int cb = array.get(arrB, i)
            if cb <= afterBar
                break
            float cp = array.get(arrP, i)
            if (wantLow ? cp <= beyond : cp >= beyond) and (na(p) or (wantLow ? cp < p : cp > p))
                p := cp
                b := cb
    [p, b]


f_swpConf(array<int> arrB, int swpB) =>
    bool ok = false
    int  n  = array.size(arrB)
    if not na(swpB) and n > 0
        for i = n - 1 to 0
            int cb = array.get(arrB, i)
            if cb == swpB
                ok := true
                break
            if cb < swpB
                break
    ok

f_lastMinor(bool wantLow, int beforeBar) =>
    arrB = wantLow ? msLoB : msHiB
    arrP = wantLow ? msLoP : msHiP
    float p = na
    int   b = na
    int   n = array.size(arrB)
    if n > 0
        for i = n - 1 to 0
            if array.get(arrB, i) < beforeBar
                p := array.get(arrP, i)
                b := array.get(arrB, i)
                break
    [p, b]

f_idmMark(SLine s) =>
    bool inWin = (last_bar_index - bar_index) <= histBars
    if showIdm and inWin and not na(s.idmP) and not na(s.idmB) and (bar_index - s.idmB) <= histBars
        bool isCeil = s.sid == 0 or s.sid == 2
        label.new(s.idmB, s.idmP, idmTxt, color = isCeil ? idmUpC : idmDnC, textcolor = idmTc, style = isCeil ? label.style_label_up : label.style_label_down, size = tsize(idmSz))

method armIdm(SLine s) =>
    if not na(s)
        bool isCeil = s.sid == 0 or s.sid == 2
        [ip, ib] = f_lastMinor(isCeil, s.x1)
        s.idmP   := ip
        s.idmB   := ib
        s.idmHit := false
        if not na(ip)
            [xp, xb] = f_retExtreme(s.x1, not isCeil)
            bool takenPast = not na(xp) and (isCeil ? xp < ip : xp > ip)
            bool takenNow  = isCeil ? low < ip : high > ip
            if takenPast or takenNow
                s.idmHit := true
                f_idmMark(s)

f_intArmIdm(int pivB, bool isCeil) =>
    [ip, ib] = f_lastMinor(isCeil, pivB)
    bool hit = false
    if not na(ip)
        [xp, xb] = f_retExtreme(pivB, not isCeil)
        bool takenPast = not na(xp) and (isCeil ? xp < ip : xp > ip)
        bool takenNow  = isCeil ? low < ip : high > ip
        hit := takenPast or takenNow
    [ip, hit]

f_hLine(line ln, label lb, bool on, float p, int b, string txt, color lc, color tc, string ls, int w, string ts, bool showT, string tv, string th) =>
    line  _l = ln
    label _b = lb
    if not on or na(p) or na(b)
        if not na(_l)
            line.delete(_l)
            _l := na
        if not na(_b)
            label.delete(_b)
            _b := na
    else
        int _x1 = b
        int _hMin = last_bar_index - histBars
        if _x1 < _hMin
            _x1 := _hMin
        if _x1 < 0
            _x1 := 0
        if _x1 > bar_index
            _x1 := bar_index
        if na(_l)
            _l := line.new(_x1, p, bar_index, p, color = lc, width = w, style = lstyle(ls), extend = extend.right)
        else
            line.set_xy1(_l, _x1, p)
            line.set_xy2(_l, bar_index, p)
            line.set_color(_l, lc)
            line.set_width(_l, w)
            line.set_style(_l, lstyle(ls))
        if showT
            int _lx = th == "Left" ? _x1 : th == "Right" ? bar_index : int(math.avg(_x1, bar_index))
            if na(_b)
                _b := label.new(_lx, p, txt, color = color(na), textcolor = tc, style = lblStyle(tv), size = tsize(ts))
            else
                label.set_xy(_b, _lx, p)
                label.set_text(_b, txt)
                label.set_textcolor(_b, tc)
                label.set_style(_b, lblStyle(tv))
                label.set_size(_b, tsize(ts))
        else
            if not na(_b)
                label.delete(_b)
                _b := na
    [_l, _b]

f_ipStyle(bool isBos) =>
    if isBos
        [ip1_lc, ip1_tc, ip1_ls, ip1_w, ip1_txt, ip1_ts, ip1_show, ip1_tv, ip1_th]
    else
        [ip2_lc, ip2_tc, ip2_ls, ip2_w, ip2_txt, ip2_ts, ip2_show, ip2_tv, ip2_th]

f_npStyle(bool isBos) =>
    if isBos
        [np1_lc, np1_tc, np1_ls, np1_w, np1_txt, np1_ts, np1_show, np1_tv, np1_th]
    else
        [np2_lc, np2_tc, np2_ls, np2_w, np2_txt, np2_ts, np2_show, np2_tv, np2_th]

f_eqLvl(float a, float b, float pct) =>
    float l = na
    if not na(a) and not na(b)
        float _lo = math.min(a, b)
        float _hi = math.max(a, b)
        l := _lo + (_hi - _lo) * pct / 100.0
    l

f_eqBar(int b1, int b2, string mode) =>
    int r = na
    if not na(b1) and not na(b2)
        int _older = b1 < b2 ? b1 : b2
        int _newer = b1 > b2 ? b1 : b2
        r := mode == "Newer level" ? _newer : mode == "Current bar" ? bar_index : _older
    r

f_eqTxt(string t, bool app, float pct) =>
    string s = t
    if app
        string _p = str.tostring(pct, "#.##") + "%"
        s := t == "" ? _p : t + " " + _p
    s

f_capMarks(array<line> la, array<label> lb, int capL, int capB) =>
    while array.size(la) > capL
        line.delete(array.shift(la))
    while array.size(lb) > capB
        label.delete(array.shift(lb))
