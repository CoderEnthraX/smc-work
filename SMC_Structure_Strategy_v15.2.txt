type SwingTracker
    bool  isHigh
    float potential
    int   potentialBar
    int   count = 0
    float refLevel
    bool  swept = false

method update(SwingTracker t, float swpLvl) =>
    float confP = na
    int   confB = na
    bool bearC  = close < open
    bool bullC  = close > open
    float _pbBd = math.abs(close - open)
    float _pbWk = math.max(high - math.max(open, close), math.min(open, close) - low)
    bool  _pbQ  = not (pbBodyOn or pbBodyRef) or _pbBd > _pbWk
    bool pb     = (t.isHigh ? bearC : bullC) and _pbQ
    float _pbRf = pbBodyRef ? (t.isHigh ? math.min(open, close) : math.max(open, close)) : (t.isHigh ? low : high)
    bool newExt = na(t.potential) or (t.isHigh ? high >= t.potential : low <= t.potential)
    if newExt
        t.potential    := t.isHigh ? high : low
        t.potentialBar := bar_index
        t.count        := pb ? 1 : 0
        t.refLevel     := pb ? _pbRf : na
        t.swept        := false
    else
        if not na(swpLvl) and (t.isHigh ? low < swpLvl : high > swpLvl)
            t.swept := true
        if pb
            if t.count == 0
                t.count    := 1
                t.refLevel := _pbRf
            else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel) and t.swept
                confP := t.potential
                confB := t.potentialBar
                t.potential    := t.isHigh ? high : low
                t.potentialBar := bar_index
                t.count        := 1
                t.refLevel     := _pbRf
                t.swept        := false
            else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
                t.count    := 2
                t.refLevel := _pbRf
            else if t.count == 2 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
                confP := t.potential
                confB := t.potentialBar
                t.potential    := t.isHigh ? high : low
                t.potentialBar := bar_index
                t.count        := 1
                t.refLevel     := _pbRf
                t.swept        := false
    [confP, confB]

f_retExt(int fromBar, bool findHigh, bool inc) =>
    int sz = array.size(bufLo)
    int e  = inc ? sz - 1 : sz - 2
    float p  = na
    int   b  = na
    if e >= 0
        array<float> src = findHigh ? bufHi : bufLo
        for j = math.max(0, math.min(fromBar - bar_index + sz, e)) to e
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := bar_index - sz + 1 + j
    [p, b]


method rearm(SwingTracker t, float p, int b) =>
    t.potential    := p
    t.potentialBar := b
    int  sz  = array.size(bufHi)
    int  idx = b - bar_index + sz - 1
    bool ok  = not na(b) and idx >= 0 and idx <= sz - 1
    int  dir = ok ? array.get(bufDir, idx) : 0
    bool pb  = t.isHigh ? dir == -1 : dir == 1
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? array.get(bufLo, idx) : array.get(bufHi, idx)) : na
    t.swept    := false

method armHere(SwingTracker t) =>
    t.potential    := t.isHigh ? high : low
    t.potentialBar := bar_index
    bool pb = t.isHigh ? close < open : close > open
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? low : high) : na
    t.swept    := false

    if not na(swpB) and n > 0
        for i = n - 1 to 0
            int cb = array.get(arrB, i)
            if cb == swpB
                ok := true
                break
            if cb < swpB
                break
    ok

    float p = na

    int   n = array.size(arrB)
    if n > 0
        for i = n - 1 to 0
            if array.get(arrB, i) < beforeBar
                p := array.get(arrP, i)
                b := array.get(arrB, i)
                break
    [p, b]


method armIdm(SLine s) =>
    if not na(s)
        bool isCeil = s.sid == 0 or s.sid == 2
        [ip, ib] = f_lastMinor(isCeil, s.x1)
        s.idmP   := ip
        s.idmB   := ib
        s.idmHit := false
        if not na(ip)
            [xp, xb] = f_retExt(s.x1, not isCeil, false)
            bool takenPast = not na(xp) and (isCeil ? xp < ip : xp > ip)
            bool takenNow  = isCeil ? low < ip : high > ip
            if takenPast or takenNow
                s.idmHit := true

        bool takenPast = not na(xp) and (isCeil ? xp < ip : xp > ip)
        bool takenNow  = isCeil ? low < ip : high > ip
        hit := takenPast or takenNow
    [ip, hit]

// a pivot of the tier's own swing tracker becomes the level if it has none (d = 1 the ceiling, -1 the floor)
f_tAdopt(Tier t, int d, float p, int b) =>
    TLvl l = d == 1 ? t.c : t.f
    if not na(p) and b > t.anc and (close - p) * d < 0 and na(l.p)
        l.lvSet(p, b, d == 1)

// at a new anchor: the first stored pivot after it that the close has not passed
f_tSeed(Tier t, int d) =>
    TLvl l = d == 1 ? t.c : t.f
    array<int>   ab = d == 1 ? pvHiB : pvLoB
    array<float> ap = d == 1 ? pvHiP : pvLoP
    if na(l.p) and array.size(ab) > 0
        for s = 0 to array.size(ab) - 1
            if array.get(ab, s) > t.anc and (close - array.get(ap, s)) * d < 0
                l.lvSet(array.get(ap, s), array.get(ab, s), d == 1)
                break

// the IDM of the level taken
f_tIdm(Tier t, int d) =>
    TLvl l = d == 1 ? t.c : t.f
    if not na(l.p) and not l.idmHit and not na(l.idmP) and ((d == 1 ? low : high) - l.idmP) * d < 0
        l.idmHit := true

// a close beyond the level = the tier's CHOCH / BOS; else its sweep and re-level
f_tBrk(Tier t, int d) =>
    TLvl l = d == 1 ? t.c : t.f
    TLvl o = d == 1 ? t.f : t.c
    float x = d == 1 ? high : low
    float y = d == 1 ? low : high
    if not na(l.p) and not na(l.born) and bar_index > l.born
        if (close - l.p) * d > 0 and (t.tr != d or not altMode or l.idmHit or na(l.idmP))
            bool isCho = t.tr != d
            t.ev := true
            f_tMark(t, l.p, l.b, isCho)
            [rp, rb] = f_retExt(l.b, d == -1, true)
            if not na(rp)
                o.lvSet(rp, rb, d == -1)
            if t.st and not isCho and l.idmHit and not na(rb) and (last_bar_index - bar_index) <= 500 and f_xNear(rb)
                f_addLb(t.lb, t.cnt, label.new(rb, na, "+", yloc = d == 1 ? yloc.belowbar : yloc.abovebar, style = label.style_none, textcolor = color.gray, size = size.small))
            t.tr := d
            l.lvClr()
        else
            if (x - l.p) * d > 0
                if na(l.ref)
                    [rp, rb] = f_retExt(l.b, d == -1, false)
                    l.ref := rp
                if na(l.swp) or (x - l.swp) * d > 0
                    l.swp  := x
                    l.swpB := bar_index
                    l.swpT := false
                if altMode and not l.idmHit and not na(l.idmP) and (close - l.p) * d > 0 and not na(l.swp)
                    l.lvSet(l.swp, l.swpB, d == 1)
                    l.swpT := false
            if not na(l.swpB) and l.swpB == bar_index and t.tr == d and not na(l.ref) and (y - l.ref) * d < 0
                l.swpT := true
            if t.tr == d and not na(l.swp) and not na(l.ref) and (y - l.ref) * d < 0 and (not l.swpT or f_swpConf(d == 1 ? pvHiB : pvLoB, l.swpB))
                l.lvSet(l.swp, l.b, d == 1)

// one bar of a tier: rst = the tier above made a CHOCH / BOS (all levels and marks go), anc = the bar the tier starts after
f_tier(Tier t, bool rst, int anc, float hp, int hb, float lp, int lq) =>
    t.ev := false
    if rst
        t.tr  := 0
        t.anc := na
        t.c.lvClr()
        t.f.lvClr()
        f_cutAll(t.ln, t.lb, t.cnt)
    if na(t.anc) and not na(anc)
        t.anc := anc
        f_tSeed(t, 1)
        f_tSeed(t, -1)
    if not rst and not na(t.anc)
        f_tAdopt(t, 1, hp, hb)
        f_tAdopt(t, -1, lp, lq)
        f_tIdm(t, 1)
        f_tIdm(t, -1)
        f_tBrk(t, 1)
        f_tBrk(t, -1)

// US market holiday on a New York date, "" if none. gf: count Good Friday (markets shut, government offices open)
f_usHol(int y, int m, int d, bool gf) =>
    string r = ""
    if m == 1 and ((d == 1 and f_dow(y, 1, 1) != 1 and f_dow(y, 1, 1) != 7) or (d == 2 and f_dow(y, 1, 1) == 1))
        r := "New Year"
    else if m == 1 and d == f_nthDow(y, 1, dayofweek.monday, 3)
        r := "Martin Luther King Day"
    else if m == 2 and d == f_nthDow(y, 2, dayofweek.monday, 3)
        r := "Presidents' Day"
    else if m == 5 and d == f_lastDow(y, 5, dayofweek.monday)
        r := "Memorial Day"
    else if y >= 2022 and f_obs(y, 6, 19, m, d)
        r := "Juneteenth"
    else if f_obs(y, 7, 4, m, d)
        r := "Independence Day"
    else if m == 9 and d == f_nthDow(y, 9, dayofweek.monday, 1)
        r := "Labor Day"
    else if m == 11 and d == f_nthDow(y, 11, dayofweek.thursday, 4)
        r := "Thanksgiving"
    else if f_obs(y, 12, 25, m, d)
        r := "Christmas"
    else if gf and (m == 3 or m == 4)
        // Easter Sunday (Gregorian computus); Good Friday is two days before
        int a  = y % 19
        int b  = int(y / 100)
        int c  = y % 100
        int h  = (19 * a + b - int(b / 4) - int((b - int((b + 8) / 25) + 1) / 3) + 15) % 30
        int l  = (32 + 2 * (b % 4) + 2 * int(c / 4) - h - c % 4) % 7
        int q  = int((a + 11 * h + 22 * l) / 451)
        [gy, gm, gd] = f_addD(y, int((h + l - 7 * q + 114) / 31), (h + l - 7 * q + 114) % 31 + 1, -2)
        if m == gm and d == gd
            r := "Good Friday"
    r

        rm := m3
        rd := d3
