f_hhmm(int h, int m) =>
    (h < 10 ? "0" : "") + str.tostring(h, "0") + (m < 10 ? "0" : "") + str.tostring(m, "0")

f_sesShift(string ses, int hrs) =>
    string out = ses
    if hrs != 0
        int dash = str.pos(ses, "-")
        if not na(dash) and dash >= 4 and str.length(ses) >= dash + 5
            float ahf = str.tonumber(str.substring(ses, dash - 4, dash - 2))
            float amf = str.tonumber(str.substring(ses, dash - 2, dash))
            float bhf = str.tonumber(str.substring(ses, dash + 1, dash + 3))
            float bmf = str.tonumber(str.substring(ses, dash + 3, dash + 5))
            if not na(ahf) and not na(amf) and not na(bhf) and not na(bmf)
                int ah = (int(ahf) + hrs) % 24
                int bh = (int(bhf) + hrs) % 24
                if ah < 0
                    ah += 24
                if bh < 0
                    bh += 24
                string tail = str.length(ses) > dash + 5 ? str.substring(ses, dash + 5) : ""
                out := f_hhmm(ah, int(amf)) + "-" + f_hhmm(bh, int(bmf)) + tail
    out

f_cutOld(array<line> la, array<label> lb, array<int> cnt, int cap) =>
    while array.size(la) > cap
        line.delete(array.shift(la))
        int c = array.shift(cnt)
        while c > 0 and array.size(lb) > 0
            label.delete(array.shift(lb))
            c -= 1

f_cutAll(array<line> la, array<label> lb, array<int> cnt) =>
    while array.size(la) > 0
        line.delete(array.pop(la))
    while array.size(lb) > 0
        label.delete(array.pop(lb))
    array.clear(cnt)

type SwingTracker
    bool  isHigh
    float potential
    int   potentialBar
    int   count
    float refLevel

method update(SwingTracker t) =>
    float confP = na
    int   confB = na
    bool bearC  = close < open
    bool bullC  = close > open
    bool pb     = t.isHigh ? bearC : bullC
    bool newExt = na(t.potential) or (t.isHigh ? high >= t.potential : low <= t.potential)
    if newExt
        t.potential    := t.isHigh ? high : low
        t.potentialBar := bar_index
        t.count        := pb ? 1 : 0
        t.refLevel     := pb ? (t.isHigh ? low : high) : na
    else if pb
        if t.count == 0
            t.count    := 1
            t.refLevel := t.isHigh ? low : high
        else if t.count == 1 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
            t.count    := 2
            t.refLevel := t.isHigh ? low : high
        else if t.count == 2 and (t.isHigh ? close < t.refLevel : close > t.refLevel)
            confP := t.potential
            confB := t.potentialBar
            t.potential    := t.isHigh ? high : low
            t.potentialBar := bar_index
            t.count        := 1
            t.refLevel     := t.isHigh ? low : high
    [confP, confB]

f_retExtreme(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd > 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd - 1))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd - 1
            float v = array.get(src, j)
            if na(p) or (findHigh ? v >= p : v <= p)
                p := v
                b := base + j
    [p, b]

f_retExtInc(int fromBar, bool findHigh) =>
    int sz   = array.size(bufLo)
    int jEnd = sz - 1
    float p  = na
    int   b  = na
    if jEnd >= 0
        int j0 = math.max(0, math.min(fromBar - bar_index + sz, jEnd))
        array<float> src = findHigh ? bufHi : bufLo
        int base = bar_index - sz + 1
        for j = j0 to jEnd

method rearm(SwingTracker t, float p, int b) =>
    t.potential    := p
    t.potentialBar := b
    int  sz  = array.size(bufHi)
    int  idx = b - bar_index + sz - 1
    bool ok  = not na(b) and idx >= 0 and idx <= sz - 1
    int  dir = ok ? array.get(bufDir, idx) : 0
    bool pb  = t.isHigh ? dir == -1 : dir == 1
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? array.get(bufLo, idx) : array.get(bufHi, idx)) : na

method armHere(SwingTracker t) =>
    t.potential    := t.isHigh ? high : low
    t.potentialBar := bar_index
    bool pb = t.isHigh ? close < open : close > open
    t.count    := pb ? 1 : 0
    t.refLevel := pb ? (t.isHigh ? low : high) : na

f_isSwing(int pb, bool isHigh, int k) =>
    int  sz  = array.size(bufHi)
    int  idx = pb - bar_index + sz - 1
    bool ok  = not na(pb) and idx - k >= 0 and idx <= sz - 1
    if ok
        float p = isHigh ? array.get(bufHi, idx) : array.get(bufLo, idx)
        for j = 1 to k
            float v = isHigh ? array.get(bufHi, idx - j) : array.get(bufLo, idx - j)
            if isHigh ? v > p : v < p
                ok := false
    ok

    float p = na

            float cp = array.get(arrP, i)

            if cb < swpB

method armIdm(SLine s) =>
    if not na(s)
        bool isCeil = s.sid == 0 or s.sid == 2
        [ip, ib] = f_lastMinor(isCeil, s.x1)
        s.idmP   := ip
        s.idmB   := ib
        s.idmHit := false
        if not na(ip)
            [xp, xb] = f_retExtreme(s.x1, not isCeil)
            bool takenPast = not na(xp) and (isCeil ? xp < ip : xp > ip)
            bool takenNow  = isCeil ? low < ip : high > ip
            if takenPast or takenNow
                s.idmHit := true
                f_idmMark(s)

        bool takenNow  = isCeil ? low < ip : high > ip
        hit := takenPast or takenNow
    [ip, hit]

method feed(VPCol c) =>
    if c.on
        c.hi := math.max(c.hi, high)
        c.lo := math.min(c.lo, low)
        int n = array.size(ltfH)
        if n > 0
            array.concat(c.prH, ltfH)
            array.concat(c.prL, ltfL)
            array.concat(c.vo,  ltfV)
            for i = 0 to n - 1
                array.push(c.up,  array.get(ltfC, i) >= array.get(ltfO, i))
        else
            array.push(c.prH, high)
            array.push(c.prL, low)
            array.push(c.vo,  nz(volume))
            array.push(c.up,  close >= open)

f_render(VPCol c, int x2, VPStyle st, DrawQueue dq, int maxProf, bool rightSide, int rightOff) =>
    int n = array.size(c.prH)
    if n > 0 and c.hi > c.lo
        float rngAll = c.hi - c.lo
        float tick   = syminfo.mintick
        float tprF   = rngAll / vpRows / tick
        int   tprA   = math.max(1, int(math.floor(tprF)))
        int   tprB   = math.max(1, int(math.ceil(tprF)))
        int   rowsA  = int(math.ceil(rngAll / (tprA * tick) - 1e-9))
        int   rowsB  = int(math.ceil(rngAll / (tprB * tick) - 1e-9))
        int   tpr    = math.abs(rowsA - vpRows) <= math.abs(rowsB - vpRows) ? tprA : tprB
        float step   = tpr * tick
        int   rows   = math.min(100, math.max(1, int(math.ceil(rngAll / step - 1e-9))))
        binUp = array.new_float(rows, 0.0)
        binDn = array.new_float(rows, 0.0)
        float lo0  = c.lo
        int   rMax = rows - 1
        for i = 0 to n - 1
            float v = nz(array.get(c.vo, i))
            if v > 0
                float cH  = array.get(c.prH, i)
                float cL  = array.get(c.prL, i)
                bool  isU = array.get(c.up, i)
                int loBin = int(math.max(0, math.min(rMax, math.floor((cL - lo0) / step))))
                int hiBin = int(math.max(0, math.min(rMax, math.floor((cH - lo0) / step))))
                if hiBin < loBin
                    hiBin := loBin
                float rng = cH - cL
                for b = loBin to hiBin
                    float share = v
                    if rng > 0
                        float eb = lo0 + step * b
                        float et = eb + step
                        share := v * (math.min(cH, et) - math.max(cL, eb)) / rng
                    if share > 0
                        if isU
                            array.set(binUp, b, array.get(binUp, b) + share)
                        else
                            array.set(binDn, b, array.get(binDn, b) + share)
        binv = array.new_float(rows, 0.0)
        float tot  = 0.0
        float mx   = 0.0
        int   pocI = 0
        float ctr  = rMax / 2.0
        for r = 0 to rMax
            float vr = array.get(binUp, r) + array.get(binDn, r)
            array.set(binv, r, vr)
            tot += vr
            if vr > mx
                mx   := vr
                pocI := r
            else if vr == mx and math.abs(r - ctr) < math.abs(pocI - ctr)
                pocI := r
        if mx > 0
            int vaU = pocI
            int vaD = pocI
            float remain = tot * vpVa / 100 - mx
            bool vaGo = remain > 0
            while vaGo and (vaU < rows - 1 or vaD > 0)
                float vUp = vaU < rows - 1 ? array.get(binv, vaU + 1) : na
                float vDn = vaD > 0 ? array.get(binv, vaD - 1) : na
                bool pickUp = false
                if na(vDn)
                    pickUp := true
                else if na(vUp)
                    pickUp := false
                else if vUp != vDn
                    pickUp := vUp > vDn
                else
                    int dU = vaU + 1 - pocI
                    int dD = pocI - (vaD - 1)
                    pickUp := dU <= dD
                float vSel = pickUp ? vUp : vDn
                if vSel > remain
                    vaGo := false
                else
                    remain -= vSel
                    if pickUp
                        vaU += 1
                    else
                        vaD -= 1
            int span = math.max(1, x2 - c.x1)
            int wMax = math.max(1, int(math.round(span * vpWid / 100.0)))
            if rightSide
                wMax := math.max(1, math.min(wMax, 480 - rightOff))
            int anchor = rightSide ? x2 + rightOff + wMax : c.x1

            int bPushed = 0
            bool  upDn = st.volMode == "Up/Down"
            color sUpC = st.upC
            color sDnC = st.dnC
            color sVaU = st.vaUpC
            color sVaD = st.vaDnC
            float hiV  = c.hi
            for r = 0 to rMax
                float v = array.get(binv, r)
                if v > 0
                    int w = math.max(1, int(math.round(wMax * v / mx)))
                    bool inVA = r >= vaD and r <= vaU
                    float top = math.min(hiV, lo0 + step * (r + 1))
                    float bot = lo0 + step * r
                    if upDn
                        float uv = array.get(binUp, r)
                        int uw = v > 0 ? int(math.round(w * uv / v)) : 0
                        int dw = w - uw
                        color ucol = inVA ? sVaU : sUpC
                        color dcol = inVA ? sVaD : sDnC
                        if rightSide
                            if dw > 0
                                array.push(dq.bq, box.new(anchor - dw, top, anchor, bot, border_color = color(na), bgcolor = dcol))
                                bPushed += 1
                            if uw > 0
                                array.push(dq.bq, box.new(anchor - w, top, anchor - dw, bot, border_color = color(na), bgcolor = ucol))
                                bPushed += 1
                        else
                            if uw > 0
                                array.push(dq.bq, box.new(anchor, top, anchor + uw, bot, border_color = color(na), bgcolor = ucol))
                                bPushed += 1
                            if dw > 0
                                array.push(dq.bq, box.new(anchor + uw, top, anchor + w, bot, border_color = color(na), bgcolor = dcol))
                                bPushed += 1
                    else
                        color col = inVA ? sVaU : sUpC
                        if rightSide
                            array.push(dq.bq, box.new(anchor - w, top, anchor, bot, border_color = color(na), bgcolor = col))
                            bPushed += 1
                        else
                            array.push(dq.bq, box.new(anchor, top, anchor + w, bot, border_color = color(na), bgcolor = col))
                            bPushed += 1

            int lPushed = 0
            float pocB = lo0 + step * pocI
            float pocP = (pocB + math.min(hiV, pocB + step)) / 2
            float vahP = math.min(hiV, lo0 + step * (vaU + 1))
            float valP = lo0 + step * vaD
            int lineR = rightSide ? anchor : x2
            if st.pocOn
                array.push(dq.lq, line.new(c.x1, pocP, lineR, pocP, color = st.pocC, width = st.pocW, style = lstyle(st.pocS), extend = st.extLines ? extend.right : extend.none))
                lPushed += 1
            if st.vahOn
                array.push(dq.lq, line.new(c.x1, vahP, lineR, vahP, color = st.vahC, width = st.vahW, style = lstyle(st.vahS), extend = st.extLines ? extend.right : extend.none))
                lPushed += 1
            if st.valOn
                array.push(dq.lq, line.new(c.x1, valP, lineR, valP, color = st.valC, width = st.valW, style = lstyle(st.valS), extend = st.extLines ? extend.right : extend.none))
                lPushed += 1

            array.push(dq.bCnt, bPushed)
            array.push(dq.lCnt, lPushed)
            array.push(dq.dTag, dayNum)

            while array.size(dq.bCnt) > maxProf
                int remB = array.shift(dq.bCnt)
                if array.size(dq.dTag) > 0
                    array.shift(dq.dTag)
                if remB > 0
                    for i = 1 to remB
                        if array.size(dq.bq) > 0
                            box.delete(array.shift(dq.bq))

            while array.size(dq.lCnt) > maxProf
                int remL = array.shift(dq.lCnt)
                if remL > 0
                    for i = 1 to remL
                        if array.size(dq.lq) > 0
                            line.delete(array.shift(dq.lq))

f_wipe(DrawQueue dq) =>
    while array.size(dq.bq) > 0
        box.delete(array.shift(dq.bq))
    while array.size(dq.lq) > 0
        line.delete(array.shift(dq.lq))
    array.clear(dq.bCnt)
    array.clear(dq.lCnt)
    array.clear(dq.dTag)

f_kzbox(VPCol c, int x2, color fillC, color brdC, string brdS, int brdW, string txt, string ts, color tc, string tHal, string tVal, array<box> q, array<label> lq, array<int> qd, int keep) =>
    hA = tHal == "Left" ? text.align_left : tHal == "Right" ? text.align_right : text.align_center
    if tVal == "Outside Top" or tVal == "Outside Bottom"
        array.push(q, box.new(c.x1, c.hi, x2, c.lo, border_color = brdC, border_width = brdW, border_style = lstyle(brdS), bgcolor = fillC))
        int lx = tHal == "Left" ? c.x1 : tHal == "Right" ? x2 : int((c.x1 + x2) / 2)
        string lSt = tVal == "Outside Top" ? label.style_label_down : label.style_label_up
        float ly = tVal == "Outside Top" ? c.hi : c.lo
        array.push(lq, label.new(lx, ly, text = txt, color = color(na), textcolor = tc, style = lSt, size = tsize(ts), textalign = text.align_center))
    else
        vA = tVal == "Bottom" ? text.align_bottom : tVal == "Middle" ? text.align_center : text.align_top
        array.push(q, box.new(c.x1, c.hi, x2, c.lo, border_color = brdC, border_width = brdW, border_style = lstyle(brdS), bgcolor = fillC, text = txt, text_size = tsize(ts), text_color = tc, text_halign = hA, text_valign = vA))
        array.push(lq, label.new(bar_index, close, text="", color=color(na), textcolor=color(na)))
    array.push(qd, dayNum)

f_kzWipeLive(array<box> q, array<label> lq, array<int> qd) =>
    while array.size(q) > 0
        box.delete(array.pop(q))
    while array.size(lq) > 0
        label.delete(array.pop(lq))
    array.clear(qd)

f_sesPrune(DrawQueue dq, array<box> kq, array<label> klq, array<int> kqd) =>
    while array.size(dq.dTag) > 0 and array.get(dq.dTag, 0) <= dayNum - vpSesDays
        array.shift(dq.dTag)
        int remB = array.shift(dq.bCnt)
        if remB > 0
            for i = 1 to remB
                if array.size(dq.bq) > 0
                    box.delete(array.shift(dq.bq))
        int remL = array.shift(dq.lCnt)
        if remL > 0
            for i = 1 to remL
                if array.size(dq.lq) > 0
                    line.delete(array.shift(dq.lq))
    while array.size(kqd) > 0 and array.get(kqd, 0) <= dayNum - vpSesDays
        array.shift(kqd)
        if array.size(kq) > 0
            box.delete(array.shift(kq))
        if array.size(klq) > 0
            label.delete(array.shift(klq))

f_hLine(line ln, label lb, bool on, float p, int b, string txt, color lc, color tc, string ls, int w, string ts, bool showT, string tv, string th) =>
    line  _l = ln
    label _b = lb
    if not on or na(p) or na(b)
        if not na(_l)
            line.delete(_l)
            _l := na
        if not na(_b)
            label.delete(_b)
            _b := na
    else
        int _x1 = b
        int _hMin = last_bar_index - histBars
        if _x1 < _hMin
            _x1 := _hMin
        if _x1 < 0
            _x1 := 0
        if _x1 > bar_index
            _x1 := bar_index
        if na(_l)
            _l := line.new(_x1, p, bar_index, p, color = lc, width = w, style = lstyle(ls), extend = extend.right)
        else
            line.set_xy1(_l, _x1, p)
            line.set_xy2(_l, bar_index, p)
            line.set_color(_l, lc)
            line.set_width(_l, w)
            line.set_style(_l, lstyle(ls))
        if showT
            int _lx = th == "Left" ? _x1 : th == "Right" ? bar_index : int(math.avg(_x1, bar_index))
            if na(_b)
                _b := label.new(_lx, p, txt, color = color(na), textcolor = tc, style = lblStyle(tv), size = tsize(ts))
            else
                label.set_xy(_b, _lx, p)
                label.set_text(_b, txt)
                label.set_textcolor(_b, tc)
                label.set_style(_b, lblStyle(tv))
                label.set_size(_b, tsize(ts))
        else
            if not na(_b)
                label.delete(_b)
                _b := na
    [_l, _b]

f_eqLvl(float a, float b, float pct) =>
    float l = na
    if not na(a) and not na(b)
        float _lo = math.min(a, b)
        float _hi = math.max(a, b)
        l := _lo + (_hi - _lo) * pct / 100.0
    l

    r
